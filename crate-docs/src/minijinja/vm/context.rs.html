15pub(crate) struct Frame<'env> {
16    pub(crate) locals: Locals<'env>,
17    pub(crate) ctx: Value,
18    pub(crate) current_loop: Option<LoopState>,
19
20    // normally a frame does not carry a closure, but it can when a macro is
21    // declared.  Once that happens, all writes to the frames locals are also
22    // duplicated into the closure.  Macros declared on that level, then share
23    // the closure object to enclose the parent values.  This emulates the
24    // behavior of closures in Jinja2.
25    #[cfg(feature = "macros")]
26    pub(crate) closure: Option<ClosureId>,
27    #[cfg(feature = "macros")]
28    pub(crate) closure_context: Option<ClosureId>,
29}

57#[cfg(feature = "internal_debug")]
58impl fmt::Debug for Frame<'_> {
59    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
60        let mut m = f.debug_map();
61        m.entry(&"locals", &self.locals);
62        if let Some(LoopState {
63            object: ref controller,
64            ..
65        }) = self.current_loop
66        {
67            m.entry(&"loop", controller);
68        }
69        if !self.ctx.is_undefined() {
70            m.entry(&"ctx", &self.ctx);
71        }
72        m.finish()
73    }
74}

146impl fmt::Debug for ContextDebug<'_, '_> {
147    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
148        let vars = self.context.known_variables(
149            #[cfg(feature = "macros")]
150            self.closures,
151            false,
152        );
153        f.debug_map()
154            .entries(vars.into_iter().map(|key| {
155                let value = self
156                    .context
157                    .load(
158                        #[cfg(feature = "macros")]
159                        self.closures,
160                        &key,
161                    )
162                    .unwrap_or_default();
163                (key, value)
164            }))
165            .finish()
166    }
167}

217    /// Stores a variable in the context.
218    pub fn store(
219        &mut self,
220        #[cfg(feature = "macros")] closures: &mut [Closure<'env>],
221        key: &'env str,
222        value: Value,
223    ) {
224        let top = self.stack.last_mut().unwrap();
225        #[cfg(feature = "macros")]
226        if let Some(closure) = top.closure {
227            closures[closure].insert(key, value.clone());
228        }
229        top.locals.insert(key, value);
230    }

232    /// Adds a value to a closure if missing.
233    ///
234    /// All macros declared on a certain level reuse the same closure.  This is
235    /// done to emulate the behavior of how scopes work in Jinja2 in Python.
236    #[cfg(feature = "macros")]
237    pub fn enclose(&self, closures: &mut [Closure<'env>], key: &'env str) {
238        let closure = self.stack.last().unwrap().closure.unwrap();
239        if !closures[closure].contains_key(key) {
240            let value = self.load(closures, key).unwrap_or(Value::UNDEFINED);
241            closures[closure].insert(key, value);
242        }
243    }

251    /// Temporarily takes the closure.
252    ///
253    /// This is done because includes are in the same scope as the module that
254    /// triggers the import, but we do not want to allow closures to be modified
255    /// from another file as this would be very confusing.
256    ///
257    /// This means that if you override a variable referenced by a macro after
258    /// including in the parent template, it will not override the value seen by
259    /// the macro.

280    /// Looks up a variable in the context.
281    pub fn load(
282        &self,
283        #[cfg(feature = "macros")] closures: &[Closure<'env>],
284        key: &str,
285    ) -> Option<Value> {
286        for frame in self.stack.iter().rev() {
287            // look at locals first
288            if let Some(value) = frame.locals.get(key) {
289                return Some(value.clone());
290            }
291
292            // if we are a loop, check if we are looking up the special loop var.
293            if let Some(ref l) = frame.current_loop {
294                if l.with_loop_var && key == "loop" {
295                    return Some(Value::from_dyn_object(l.object.clone()));
296                }
297            }
298
299            #[cfg(feature = "macros")]
300            if let Some(closure) = frame.closure_context {
301                if let Some(value) = closures.get(closure).and_then(|closure| closure.get(key)) {
302                    return Some(value.clone());
303                }
304            }
305
306            // perform a fast lookup.  This one will not produce errors if the
307            // context is undefined or of the wrong type.
308            if let Some(rv) = frame.ctx.get_attr_fast(key) {
309                return Some(rv);
310            }
311        }
312
313        self.env.get_global(key)
314    }

316    /// Returns an iterable of all declared variables.
317    pub fn known_variables(
318        &self,
319        #[cfg(feature = "macros")] closures: &[Closure<'env>],
320        with_globals: bool,
321    ) -> BTreeSet<Cow<'_, str>> {
322        let mut seen = BTreeSet::<Cow<'_, str>>::new();
323        for frame in self.stack.iter().rev() {
324            for key in frame.locals.keys() {
325                seen.insert(Cow::Borrowed(*key));
326            }
327
328            if let Some(ref l) = frame.current_loop {
329                if l.with_loop_var {
330                    seen.insert(Cow::Borrowed("loop"));
331                }
332            }
333
334            #[cfg(feature = "macros")]
335            if let Some(closure) = frame.closure_context {
336                if let Some(closure) = closures.get(closure) {
337                    seen.extend(closure.keys().map(|key| Cow::Borrowed(*key)));
338                }
339            }
340
341            if let Ok(iter) = frame.ctx.try_iter() {
342                for key in iter {
343                    if let Some(str_key) = key.as_str() {
344                        if !seen.contains(&Cow::Borrowed(str_key))
345                            && frame.ctx.get_item(&key).is_ok()
346                        {
347                            seen.insert(Cow::Owned(str_key.to_owned()));
348                        }
349                    }
350                }
351            }
352        }
353        if with_globals {
354            seen.extend(self.env.globals().map(|x| Cow::Borrowed(x.0)));
355        }
356        seen
357    }

359    /// Pushes a new layer.
360    pub fn push_frame(&mut self, layer: Frame<'env>) -> Result<(), Error> {
361        self.stack.push(layer);
362        if let Err(err) = self.check_depth() {
363            self.stack.pop();
364            return Err(err);
365        }
366        Ok(())
367    }

396    pub fn next_loop_item(&mut self) -> Option<Value> {
397        let frame = self
398            .stack
399            .iter_mut()
400            .rev()
401            .find(|x| x.current_loop.is_some())?;
402        let item = frame.current_loop.as_mut()?.next();
403        if item.is_some() {
404            frame.locals.clear();
405        }
406        item
407    }

432    /// Increase the stack depth.
433    #[cfg(any(feature = "macros", feature = "multi_template"))]
434    pub fn incr_depth(&mut self, delta: usize) -> Result<(), Error> {
435        self.outer_stack_depth += delta;
436        if let Err(err) = self.check_depth() {
437            self.outer_stack_depth -= delta;
438            return Err(err);
439        }
440        Ok(())
441    }

443    /// Decrease the stack depth.
444    #[cfg(feature = "multi_template")]
445    pub fn decr_depth(&mut self, delta: usize) {
446        self.outer_stack_depth -= delta;
447    }

449    fn check_depth(&self) -> Result<(), Error> {
450        if self.depth() > self.recursion_limit {
451            return Err(Error::new(
452                ErrorKind::InvalidOperation,
453                "recursion limit exceeded",
454            ));
455        }
456        Ok(())
457    }
