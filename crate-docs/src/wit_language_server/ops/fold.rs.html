8/// Implement [range folding][spec].
9///
10/// [spec]: https://microsoft.github.io/language-server-protocol/specifications/lsp/3.17/specification/#textDocument_foldingRange
11#[salsa::tracked]
12#[tracing::instrument(level = "debug", skip_all)]
13pub fn folding_range(db: &dyn Db, ast: Ast) -> Vector<FoldingRange> {
14    let db = db.as_wit();
15    let root = ast.root_node(db);
16    let src = ast.src(db);
17
18    folding_range_impl(root, src)
19}
20
21fn folding_range_impl(root: Node<'_>, src: &str) -> Vector<FoldingRange> {
22    let positions = Positions::for_text(src);
23    let mut cursor = QueryCursor::new();
24    let language = tree_sitter_wit::language();
25    let query = tree_sitter::Query::new(&language, tree_sitter_wit::FOLDING_QUERY)
26        .expect("The query was invalid");
27
28    let capture_names = query.capture_names();
29
30    let mut ranges = Vector::new();
31    let mut matches = cursor.matches(&query, root, src.as_bytes());
32    while let Some(m) = matches.next() {
33        for capture in m.captures() {
34            let kind = match capture_names[capture.index as usize] {
35                "comments" => Some(FoldingRangeKind::Comment),
36                "imports" => Some(FoldingRangeKind::Imports),
37                "normal" => None,
38                other => unreachable!(
39                    "Unknown query capture name, {other:?}, expected one of {capture_names:?}"
40                ),
41            };
42            ranges.push_back(node_range(capture.node, kind, &positions));
43        }
44    }
45    ranges
46}
47
48fn node_range(
49    node: Node<'_>,
50    kind: Option<FoldingRangeKind>,
51    positions: &Positions<'_>,
52) -> FoldingRange {
53    let range = positions.to_lsp(node.range());
54    FoldingRange {
55        start_line: range.start.line,
56        start_character: Some(range.start.character),
57        end_line: range.end.line,
58        end_character: Some(range.end.character),
59        kind,
60        collapsed_text: None,
61    }
62}
