10/// Show the declaration and documentation of the name under the cursor.
11pub fn hover(
12    db: &dyn Db,
13    ws: Workspace,
14    file: SourceFile,
15    point: tree_sitter::Point,
16    markdown: bool,
17) -> Option<Hover> {
18    let id = identifier_at(db, file, point)?;
19    let location = symbol_definition(db, ws, file, id)?;
20    let target = ws.lookup_by_path(db, location.filename.raw_path(db))?;
21    let ast = queries::parse(db, target);
22    let node = ast
23        .root_node(db)
24        .named_descendant_for_byte_range(location.range.start_byte, location.range.end_byte)?;
25    let src = target.contents(db);
26    let mut start = node.start_byte();
27    let mut docs = Vec::new();
28    let mut cursor = node.walk();
29    for attribute in node.children_by_field_name("attributes", &mut cursor) {
30        if let Some(doc) = ast::Attribute::cast(attribute).and_then(|attr| attr.doc_comment()) {
31            let text = doc.utf8_text(src).strip_prefix("///")?;
32            docs.push(
33                text.strip_prefix(' ')
34                    .unwrap_or(text)
35                    .trim_end_matches(['\r', '\n']),
36            );
37        }
38        start = attribute.end_byte();
39    }
40    let declaration = declaration_text(node, src, start);
41    let docs = docs.join("\n");
42    let mut value = if markdown {
43        // A declaration may contain comments with Markdown fences.
44        let fence = "`".repeat(
45            declaration
46                .split(|c| c != '`')
47                .map(str::len)
48                .max()
49                .unwrap_or(0)
50                .max(2)
51                + 1,
52        );
53        format!("{fence}wit\n{declaration}\n{fence}")
54    } else {
55        declaration.to_owned()
56    };
57    if !docs.is_empty() {
58        value.push_str("\n\n");
59        value.push_str(&docs);
60    }
61    Some(Hover {
62        contents: HoverContents::Markup(MarkupContent {
63            kind: if markdown {
64                MarkupKind::Markdown
65            } else {
66                MarkupKind::PlainText
67            },
68            value,
69        }),
70        range: Some(crate::utils::Positions::new(db, file).to_lsp(id.range())),
71    })
72}

74fn declaration_text(node: tree_sitter::Node<'_>, src: &str, start: usize) -> String {
75    let text = &src[start..node.end_byte()];
76    let start = start + text.len() - text.trim_start().len();
77    let body = match node.kind() {
78        "interface_item" | "world_item" => node.child_by_field_name("body"),
79        "resource_item" => ast::ResourceItem::cast(node)
80            .and_then(|resource| resource.resource_body_opt())
81            .map(|body| body.syntax()),
82        _ => None,
83    };
84    // Container hovers describe the container, not every declaration inside it.
85    if let Some(body) = body {
86        let header = src[start..body.start_byte()].trim_end();
87        let contents = if body.named_child_count() == 0 {
88            "{}"
89        } else {
90            "{ … }"
91        };
92        return format!("{header} {contents}");
93    }
94
95    let line_start = src[..start].rfind('\n').map_or(0, |offset| offset + 1);
96    let prefix = &src[line_start..start];
97    let indent = if prefix.bytes().all(|b| matches!(b, b' ' | b'\t')) {
98        prefix
99    } else {
100        ""
101    };
102    src[start..node.end_byte()]
103        .lines()
104        .enumerate()
105        .map(|(index, line)| {
106            if index == 0 {
107                line
108            } else {
109                line.strip_prefix(indent).unwrap_or(line)
110            }
111        })
112        .collect::<Vec<_>>()
113        .join("\n")
114}

121    fn contents(source: &str, markdown: bool) -> Option<String> {
122        let offset = source.find('|').unwrap();
123        let prefix = &source[..offset];
124        let point = tree_sitter::Point::new(
125            prefix.bytes().filter(|b| *b == b'\n').count(),
126            prefix.rsplit('\n').next().unwrap().len(),
127        );
128        let db = Compiler::default();
129        let path = FilePath::new(&db, "file:///types.wit".into());
130        let file = SourceFile::new(&db, path, source.replacen('|', "", 1).into());
131        let ws = Workspace::new(&db, [(path, file)].into_iter().collect());
132        let HoverContents::Markup(content) = hover(&db, ws, file, point, markdown)?.contents else {
133            panic!("expected markup content");
134        };
135        Some(content.value)
136    }

138    #[test]
139    fn shows_declarations_and_attached_documentation() {
140        for (source, expected) in [
141            (
142                "/// An interface.\ninterface |i {}",
143                "interface i {}\n\nAn interface.",
144            ),
145            ("/// A world.\nworld |w {}", "world w {}\n\nA world."),
146            (
147                "interface i { /// Count.\ntype |count = u32; }",
148                "type count = u32;\n\nCount.",
149            ),
150            (
151                "interface i { /// A record.\nrecord |r { x: u32 } }",
152                "record r { x: u32 }\n\nA record.",
153            ),
154            (
155                "interface i { record r { /// X.\n|x: u32 } }",
156                "x: u32\n\nX.",
157            ),
158            (
159                "interface i { /// Run.\n|run: func(x: u32); }",
160                "run: func(x: u32);\n\nRun.",
161            ),
162            ("interface i { f: func(|x: u32); }", "x: u32"),
163            ("interface i { enum e { /// Value.\n|v } }", "v\n\nValue."),
164            ("interface i { flags f { |v } }", "v"),
165            ("interface i { variant v { |some(u32) } }", "some(u32)"),
166            ("interface i { resource |r; }", "resource r;"),
167            ("interface i { resource r { |f: func(); } }", "f: func();"),
168            (
169                "interface i { resource r { |f: static func(); } }",
170                "f: static func();",
171            ),
172            ("world w { import |f: func(); }", "import f: func();"),
173            ("world w { export |f: func(); }", "export f: func();"),
174        ] {
175            assert_eq!(
176                contents(source, false).as_deref(),
177                Some(expected),
178                "{source}"
179            );
180        }
181    }

183    #[test]
184    fn preserves_markdown_and_doc_indentation_without_ordinary_comments() {
185        let source = "interface i { // not documentation\n/// A **count**.\r\n///\r\n///     example\r\ntype count = u32; type x = |count; }";
186        assert_eq!(
187            contents(source, true).unwrap(),
188            "```wit\ntype count = u32;\n```\n\nA **count**.\n\n    example"
189        );
190        assert_eq!(
191            contents("interface i { record |r { /* ``` */ x: u32 } }", true).unwrap(),
192            "````wit\nrecord r { /* ``` */ x: u32 }\n````"
193        );
