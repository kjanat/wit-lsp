14/// Highlight occurrences of the selected binding in this document only.
15pub fn document_highlights(
16    db: &dyn Db,
17    ws: Workspace,
18    file: SourceFile,
19    point: tree_sitter::Point,
20) -> Vec<DocumentHighlight> {
21    let Some(id) = identifier_at(db, file, point) else {
22        return Vec::new();
23    };
24    let ids = identifiers(db, file);
25    if !ids.contains(&id) || symbol_definition(db, ws, file, id).is_none() {
26        return Vec::new();
27    }
28    let Some(binding) = symbol_binding(db, ws, file, id) else {
29        return Vec::new();
30    };
31    let positions = Positions::new(db, file);
32    let mut highlights: Vec<_> = ids
33        .into_iter()
34        .filter(|id| symbol_binding(db, ws, file, *id) == Some(binding))
35        .map(|id| DocumentHighlight {
36            range: positions.to_lsp(id.range()),
37            kind: Some(DocumentHighlightKind::TEXT),
38        })
39        .collect();
40    highlights.sort_by_key(|highlight| (highlight.range.start, highlight.range.end));
41    highlights.dedup();
42    highlights
43}
