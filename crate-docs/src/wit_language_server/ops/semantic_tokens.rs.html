18pub fn semantic_tokens_legend(
19    client: Option<&SemanticTokensClientCapabilities>,
20) -> SemanticTokensLegend {
21    let token_types = vec![
22        SemanticTokenType::NAMESPACE,
23        SemanticTokenType::INTERFACE,
24        SemanticTokenType::STRUCT,
25        SemanticTokenType::CLASS,
26        SemanticTokenType::ENUM,
27        SemanticTokenType::ENUM_MEMBER,
28        SemanticTokenType::TYPE,
29        SemanticTokenType::FUNCTION,
30        SemanticTokenType::METHOD,
31        SemanticTokenType::PROPERTY,
32        SemanticTokenType::PARAMETER,
33    ];
34    SemanticTokensLegend {
35        token_types: token_types
36            .into_iter()
37            .filter(|kind| client.is_none_or(|client| client.token_types.contains(kind)))
38            .collect(),
39        token_modifiers: [SemanticTokenModifier::DECLARATION]
40            .into_iter()
41            .filter(|modifier| {
42                client.is_none_or(|client| client.token_modifiers.contains(modifier))
43            })
44            .collect(),
45    }
46}

48/// Classify identifiers through the same declarations used by navigation and hover.
49pub fn semantic_tokens(
50    db: &dyn Db,
51    ws: Workspace,
52    file: SourceFile,
53    legend: &SemanticTokensLegend,
54) -> SemanticTokens {
55    let mut ids = identifiers(db, file);
56    ids.sort_by_key(|id| id.range().start_byte);
57    let positions = Positions::new(db, file);
58    let mut data = Vec::new();
59    let mut previous = (0, 0);
60    for id in ids {
61        let declaration = declaration_for_identifier(db, file, id);
62        let kind = if declaration.is_some() {
63            id.syntax().parent().and_then(token_kind)
64        } else {
65            symbol_definition(db, ws, file, id).and_then(|location| {
66                let target = ws.lookup_by_path(db, location.filename.raw_path(db))?;
67                queries::parse(db, target)
68                    .root_node(db)
69                    .named_descendant_for_byte_range(
70                        location.range.start_byte,
71                        location.range.end_byte,
72                    )
73                    .and_then(token_kind)
74            })
75        };
76        let Some(kind) = kind else { continue };
77        let Some(token_type) = legend
78            .token_types
79            .iter()
80            .position(|supported| *supported == kind)
81            .or_else(|| {
82                [
83                    SemanticTokenType::STRUCT,
84                    SemanticTokenType::CLASS,
85                    SemanticTokenType::ENUM,
86                ]
87                .contains(&kind)
88                .then(|| {
89                    legend
90                        .token_types
91                        .iter()
92                        .position(|kind| *kind == SemanticTokenType::TYPE)
93                })
94                .flatten()
95            })
96        else {
97            continue;
98        };
99        let range = positions.to_lsp(id.range());
100        if range.start.line != range.end.line || range.start == range.end {
101            continue;
102        }
103        let modifier = (declaration.is_some() || is_alias(id))
104            && legend
105                .token_modifiers
106                .contains(&SemanticTokenModifier::DECLARATION);
107        data.push(SemanticToken {
108            delta_line: range.start.line - previous.0,
109            delta_start: if range.start.line == previous.0 {
110                range.start.character - previous.1
111            } else {
112                range.start.character
113            },
114            length: range.end.character - range.start.character,
115            token_type: token_type as u32,
116            token_modifiers_bitset: u32::from(modifier),
117        });
118        previous = (range.start.line, range.start.character);
119    }
120    SemanticTokens {
121        result_id: None,
122        data,
123    }
124}

126fn token_kind(node: tree_sitter::Node<'_>) -> Option<SemanticTokenType> {
127    if node.kind() == "named_type" {
128        return Some(SemanticTokenType::PARAMETER);
129    }
130    let (_, kind) = symbol_name(node)?;
131    Some(match kind {
132        SymbolKind::NAMESPACE | SymbolKind::PACKAGE => SemanticTokenType::NAMESPACE,
133        SymbolKind::INTERFACE => SemanticTokenType::INTERFACE,
134        SymbolKind::STRUCT => SemanticTokenType::STRUCT,
135        SymbolKind::CLASS => SemanticTokenType::CLASS,
136        SymbolKind::ENUM => SemanticTokenType::ENUM,
137        SymbolKind::ENUM_MEMBER => SemanticTokenType::ENUM_MEMBER,
138        SymbolKind::TYPE_PARAMETER => SemanticTokenType::TYPE,
139        SymbolKind::FUNCTION => SemanticTokenType::FUNCTION,
140        SymbolKind::METHOD => SemanticTokenType::METHOD,
141        SymbolKind::FIELD => SemanticTokenType::PROPERTY,
142        _ => return None,
143    })
