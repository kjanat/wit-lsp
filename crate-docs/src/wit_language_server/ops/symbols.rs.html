12/// Search declarations in all known files using a case-insensitive name fragment.
13pub fn workspace_symbols(
14    db: &dyn Db,
15    ws: Workspace,
16    query: &str,
17    supported_kinds: Option<&[SymbolKind]>,
18) -> Vec<SymbolInformation> {
19    let query = query.to_lowercase();
20    let mut results = Vec::new();
21    for (path, file) in ws.files(db) {
22        let Ok(uri) = path.raw_path(db).parse::<Url>() else {
23            continue;
24        };
25        if let DocumentSymbolResponse::Flat(symbols) =
26            document_symbols(db, file, uri, false, supported_kinds.is_some())
27        {
28            results.extend(symbols.into_iter().filter_map(|mut symbol| {
29                if !symbol.name.to_lowercase().contains(&query) {
30                    return None;
31                }
32                if let Some(kinds) = supported_kinds
33                    && !kinds.contains(&symbol.kind)
34                {
35                    let fallback = legacy_kind(symbol.kind);
36                    symbol.kind = if kinds.contains(&fallback) {
37                        fallback
38                    } else {
39                        kinds.first().copied().unwrap_or(fallback)
40                    };
41                }
42                Some(symbol)
43            }));
44        }
45    }
46    results.sort_by(|left, right| {
47        left.name
48            .cmp(&right.name)
49            .then(left.location.uri.cmp(&right.location.uri))
50            .then(left.location.range.start.cmp(&right.location.range.start))
51    });
52    results
53}

55/// Describe declarations in source order, preserving their syntactic owners.
56pub fn document_symbols(
57    db: &dyn Db,
58    file: SourceFile,
59    uri: Url,
60    hierarchical: bool,
61    extended_kinds: bool,
62) -> DocumentSymbolResponse {
63    let ast = queries::parse(db, file);
64    let symbols = collect(
65        ast.root_node(db),
66        file.contents(db),
67        &Positions::new(db, file),
68        extended_kinds,
69    );
70    if hierarchical {
71        DocumentSymbolResponse::Nested(symbols)
72    } else {
73        let mut flat = Vec::new();
74        flatten(symbols, &uri, None, &mut flat);
75        DocumentSymbolResponse::Flat(flat)
76    }
77}

79fn collect(
80    node: Node<'_>,
81    src: &str,
82    positions: &Positions<'_>,
83    extended_kinds: bool,
84) -> Vec<DocumentSymbol> {
85    let mut cursor = node.walk();
86    let children: Vec<_> = node
87        .named_children(&mut cursor)
88        .flat_map(|child| collect(child, src, positions, extended_kinds))
89        .collect();
90    if let Some((name, kind)) = symbol_name(node) {
91        if name.is_missing() || name.start_byte() == name.end_byte() {
92            return children;
93        }
94        let name_text = &src[name.byte_range()];
95        let detail = node
96            .child_by_field_name("type")
97            .or_else(|| {
98                let mut cursor = node.walk();
99                node.named_children(&mut cursor)
100                    .find(|child| child.kind() == "func_type")
101            })
102            .map(|ty| {
103                src[ty.byte_range()]
104                    .split_whitespace()
105                    .collect::<Vec<_>>()
106                    .join(" ")
107            });
108        #[allow(deprecated)] // lsp-types requires the legacy field in struct literals.
109        let symbol = DocumentSymbol {
110            name: name_text.strip_prefix('%').unwrap_or(name_text).to_owned(),
111            detail,
112            kind: if extended_kinds {
113                kind
114            } else {
115                legacy_kind(kind)
116            },
117            tags: None,
118            deprecated: None,
119            range: positions.to_lsp(node.range()),
120            selection_range: positions.to_lsp(name.range()),
121            children: (!children.is_empty()).then_some(children),
122        };
123        vec![symbol]
124    } else {
125        children
126    }
127}

129pub(super) fn symbol_name(node: Node<'_>) -> Option<(Node<'_>, SymbolKind)> {
130    let (field, kind) = match node.kind() {
131        "interface_item" => ("name", SymbolKind::INTERFACE),
132        "world_item" => ("name", SymbolKind::NAMESPACE),
133        "record_item" => ("name", SymbolKind::STRUCT),
134        "resource_item" => ("name", SymbolKind::CLASS),
135        "enum_items" | "flags_items" | "variant_items" => ("name", SymbolKind::ENUM),
136        "enum_case" | "flags_case" | "variant_case" => ("name", SymbolKind::ENUM_MEMBER),
137        "record_field" => ("name", SymbolKind::FIELD),
138        "type_item" => ("alias", SymbolKind::TYPE_PARAMETER),
139        "func_item" => (
140            "name",
141            if node
142                .parent()
143                .is_some_and(|parent| parent.kind() == "resource_method")
144            {
145                SymbolKind::METHOD
146            } else {
147                SymbolKind::FUNCTION
148            },
149        ),
150        "static_resource_method" => ("name", SymbolKind::METHOD),
151        "resource_constructor" => {
152            let mut cursor = node.walk();
153            return node
154                .children(&mut cursor)
155                .find(|child| child.kind() == "constructor")
156                .map(|name| (name, SymbolKind::CONSTRUCTOR));
157        }
158        "package_decl" => {
159            return child(node, "package_name").map(|name| (name, SymbolKind::PACKAGE));
160        }
161        "use_names_item" => {
162            return node
163                .child_by_field_name("alias")
164                .or_else(|| node.child_by_field_name("name"))
165                .map(|name| (name, SymbolKind::TYPE_PARAMETER));
166        }
167        "toplevel_use_item" => {
168            let mut cursor = node.walk();
169            let alias = node
170                .children_by_field_name("alias", &mut cursor)
171                .find(|child| child.kind() == "id");
172            return alias
173                .or_else(|| node.child_by_field_name("path"))
174                .map(|name| (name, SymbolKind::NAMESPACE));
175        }
176        "import_item" | "export_item" => {
177            let external = child(node, "extern_type");
178            let kind = if external.and_then(|ty| child(ty, "func_type")).is_some() {
179                SymbolKind::FUNCTION
180            } else {
181                SymbolKind::INTERFACE
182            };
183            return node
184                .child_by_field_name("name")
185                .or_else(|| child(node, "use_path"))
186                .map(|name| (name, kind));
187        }
188        _ => return None,
189    };
190    node.child_by_field_name(field).map(|name| (name, kind))
191}

193// Without a valueSet capability, clients only support the original 18 kinds.
194fn legacy_kind(kind: SymbolKind) -> SymbolKind {
195    match kind {
196        SymbolKind::STRUCT => SymbolKind::CLASS,
197        SymbolKind::ENUM_MEMBER => SymbolKind::CONSTANT,
198        SymbolKind::TYPE_PARAMETER => SymbolKind::VARIABLE,
199        _ => kind,
200    }
201}

203fn child<'tree>(node: Node<'tree>, kind: &str) -> Option<Node<'tree>> {
204    let mut cursor = node.walk();
205    node.named_children(&mut cursor)
206        .find(|child| child.kind() == kind)
207}

209fn flatten(
210    symbols: Vec<DocumentSymbol>,
211    uri: &Url,
212    parent: Option<&str>,
213    output: &mut Vec<SymbolInformation>,
214) {
215    for symbol in symbols {
216        #[allow(deprecated)] // lsp-types requires the legacy field in struct literals.
217        output.push(SymbolInformation {
218            name: symbol.name.clone(),
219            kind: symbol.kind,
220            tags: symbol.tags,
221            deprecated: None,
222            location: Location {
223                uri: uri.clone(),
224                range: symbol.selection_range,
225            },
226            container_name: parent.map(str::to_owned),
227        });
228        if let Some(children) = symbol.children {
229            flatten(children, uri, Some(&symbol.name), output);
230        }
231    }
232}
