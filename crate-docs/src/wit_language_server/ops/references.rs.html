11/// Find identifier occurrences that resolve to the same declaration.
12pub fn find_references(
13    db: &dyn Db,
14    ws: Workspace,
15    file: SourceFile,
16    point: tree_sitter::Point,
17    include_declaration: bool,
18) -> Vec<Location> {
19    let Some(target) =
20        identifier_at(db, file, point).and_then(|id| symbol_definition(db, ws, file, id))
21    else {
22        return Vec::new();
23    };
24    let mut references = Vec::new();
25    for (path, file) in ws.files(db) {
26        let Ok(uri) = path.raw_path(db).parse::<Url>() else {
27            continue;
28        };
29        let positions = Positions::new(db, file);
30        for id in identifiers(db, file) {
31            let declaration = declaration_for_identifier(db, file, id);
32            if !include_declaration && (declaration.is_some() || is_alias(id)) {
33                continue;
34            }
35            if declaration.or_else(|| symbol_definition(db, ws, file, id)) == Some(target) {
36                references.push(Location {
37                    uri: uri.clone(),
38                    range: positions.to_lsp(id.range()),
39                });
40            }
41        }
42    }
43
44    references.sort_by(|left, right| {
45        left.uri
46            .cmp(&right.uri)
47            .then(left.range.start.cmp(&right.range.start))
48            .then(left.range.end.cmp(&right.range.end))
49    });
50    references.dedup();
51    references
52}

54/// Identifier occurrences, excluding package segments and parser recovery placeholders.
55pub(super) fn identifiers(db: &dyn Db, file: SourceFile) -> Vec<ast::Id<'_>> {
56    let ast = queries::parse(db, file);
57    let mut ids = Vec::new();
58    let mut pending = vec![ast.root_node(db)];
59    while let Some(node) = pending.pop() {
60        if let Some(id) = ast::Id::cast(node) {
61            if node.is_missing() || node.start_byte() == node.end_byte() {
62                continue;
63            }
64            if let Some(path) = node.parent().and_then(ast::FullyQualifiedUsePath::cast)
65                && path.iter_ids().last() != Some(id)
66            {
67                continue;
68            }
69            ids.push(id);
70        } else {
71            let mut cursor = node.walk();
72            pending.extend(node.named_children(&mut cursor));
73        }
74    }
75    ids
76}

78pub(super) fn is_alias(id: ast::Id<'_>) -> bool {
79    let Some(parent) = id.syntax().parent() else {
80        return false;
81    };
82    if let Some(item) = ast::ToplevelUseItem::cast(parent) {
83        return item.iter_alias().any(|alias| alias == id);
84    }
85    matches!(parent.kind(), "use_names_item" | "include_names_item")
86        && parent.child_by_field_name("alias") == Some(id.syntax())
87}
