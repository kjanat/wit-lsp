8pub fn complete(
9    db: &dyn Db,
10    ws: Workspace,
11    file: SourceFile,
12    point: tree_sitter::Point,
13) -> Vec<Completion> {
14    let mut completions = Vec::new();
15
16    // First, add all known keywords
17    let keywords = wit_compiler::ast::KEYWORDS.iter().map(|kw| Completion {
18        text: kw.to_string(),
19        kind: CompletionKind::Keyword,
20    });
21    completions.extend(keywords);
22
23    let types = wit_compiler::queries::visible_types(db, ws, file, point);
24    completions.extend(
25        types
26            .keys()
27            .map(|name| Completion::type_name(name.raw(db).as_str())),
28    );
29
30    let ast = wit_compiler::queries::parse(db, file);
31    if let Some(ident) = ast.tree(db).ancestors(point).find_map(ast::Id::cast) {
32        let src = file.contents(db);
33        let ident = ident.utf8_text(src);
34        // The user has started writing an identifier, so limit the completions
35        // to whatever might match what they've written.
36        completions.retain(|c| c.text.starts_with(ident));
37    }
38
39    // completions.sort_by(|left, right| left.text.cmp(&right.text));
40
41    completions
42}

44#[derive(Debug, Clone, PartialEq, Eq, Hash)]
45pub struct Completion {
46    pub text: String,
47    pub kind: CompletionKind,
48}
49
50impl Completion {
51    pub fn new(text: impl Into<String>, kind: CompletionKind) -> Self {
52        Self {
53            text: text.into(),
54            kind,
55        }
56    }
57
58    fn type_name(text: impl Into<String>) -> Self {
59        Completion::new(text, CompletionKind::TypeName)
60    }
61}
62
63impl From<Completion> for CompletionItem {
64    fn from(value: Completion) -> Self {
65        CompletionItem {
66            label: value.text,
67            kind: Some(value.kind.into()),
68            ..Default::default()
69        }
70    }
71}
72
73#[derive(Debug, Copy, Clone, PartialEq, Eq, Hash)]
74pub enum CompletionKind {
75    TypeName,
76    Interface,
77    World,
78    Keyword,
79}
80
81impl From<CompletionKind> for CompletionItemKind {
82    fn from(value: CompletionKind) -> Self {
83        match value {
84            CompletionKind::TypeName => CompletionItemKind::STRUCT,
85            CompletionKind::Interface => CompletionItemKind::INTERFACE,
86            CompletionKind::World => CompletionItemKind::MODULE,
87            CompletionKind::Keyword => CompletionItemKind::KEYWORD,
88        }
89    }
90}

101    fn find_completion_location<'a>(
102        tree: &Tree,
103        src: &'a str,
104    ) -> OrdMap<Cow<'a, str>, tree_sitter::Point> {
105        tree.iter()
106            .filter_map(wit_compiler::ast::Comment::cast)
107            .map(|comment| (comment.text(src), comment.range().start_point))
108            .collect()
109    }

111    #[test]
112    fn completions_in_different_contexts() {
113        let src = r#"
114            interface foo {
115                /* first */
116
117                type x = /* second */ u32;
118            }
119
120            world bar {
121                export /* third */;
122            }
123        "#;
124        let db = Compiler::default();
125        let path = FilePath::new(&db, "complete_builtins.wit".into());
126        let file = SourceFile::new(&db, path, src.into());
127        let ws = Workspace::new(&db, OrdMap::unit(path, file));
128        let ast = wit_compiler::queries::parse(&db, file);
129        let locations = find_completion_location(ast.tree(&db), src);
130
131        let result: OrdMap<String, Vec<Completion>> = locations
132            .into_iter()
133            .map(|(k, p)| (k.to_string(), complete(&db, ws, file, p)))
134            .collect();
135
136        insta::assert_debug_snapshot!(result);
137    }

139    #[test]
140    fn completes_only_types_visible_in_the_current_file_and_scope() {
141        for scope in ["interface", "world"] {
142            let db = Compiler::default();
143            let source_path = FilePath::new(&db, "file:///source.wit".into());
144            let source_file = SourceFile::new(
145                &db,
146                source_path,
147                "interface source { type original = u32; type hidden = string; }".into(),
148            );
149            let path = FilePath::new(&db, "file:///consumer.wit".into());
150            let source = format!(
151                "{scope} consumer {{ use source.{{original as renamed}}; type local = u32; /* here */ }}"
152            );
153            let point = tree_sitter::Point::new(0, source.find("/* here */").unwrap());
154            let file = SourceFile::new(&db, path, source.into());
155            let ws = Workspace::new(
156                &db,
157                [(source_path, source_file), (path, file)]
158                    .into_iter()
159                    .collect(),
160            );
161            let mut names = complete(&db, ws, file, point)
162                .into_iter()
163                .filter(|item| item.kind == CompletionKind::TypeName)
164                .map(|item| item.text)
165                .collect::<Vec<_>>();
166            names.sort();
167            assert_eq!(names, ["local", "renamed"]);
168            for name in names {
169                let ident = wit_compiler::queries::metadata::Ident::new(&db, name.into());
170                assert!(wit_compiler::queries::resolve_type(&db, ws, file, point, ident).is_some());
171            }
172        }
173    }
