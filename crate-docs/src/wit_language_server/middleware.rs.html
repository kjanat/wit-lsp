55pub(crate) struct LoggingService<S>(pub S);

57impl<S> Service<tower_lsp::jsonrpc::Request> for LoggingService<S>
58where
59    S: Service<tower_lsp::jsonrpc::Request, Response = Option<tower_lsp::jsonrpc::Response>>,
60    S::Error: std::error::Error + 'static,
61    S::Future: Send + 'static,
62{
63    type Response = S::Response;
64    type Error = S::Error;
65    type Future =
66        Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send + 'static>>;
67
68    fn poll_ready(
69        &mut self,
70        cx: &mut std::task::Context<'_>,
71    ) -> std::task::Poll<Result<(), Self::Error>> {
72        self.0.poll_ready(cx)
73    }
74
75    fn call(&mut self, req: tower_lsp::jsonrpc::Request) -> Self::Future {
76        let id = match req.id() {
77            Some(tower_lsp::jsonrpc::Id::Number(n)) => n.to_string(),
78            Some(tower_lsp::jsonrpc::Id::String(s)) => s.clone(),
79            None | Some(tower_lsp::jsonrpc::Id::Null) => Uuid::new_v4().to_string(),
80        };
81        let method = req.method().to_string();
82
83        let fut = self.0.call(req);
84        let fut = async move {
85            let ret = fut.await;
86
87            match ret.as_ref() {
88                Ok(r) => {
89                    if let Some(err) = r.as_ref().and_then(|r| r.error()) {
90                        tracing::debug!(error = err as &dyn std::error::Error, "Returned an error");
91                    }
92                }
93                Err(err) => {
94                    tracing::error!(
95                        error = err as &dyn std::error::Error,
96                        "An error occurred while handling the request",
97                    )
98                }
99            }
100
101            ret
102        };
103
104        Box::pin(fut.instrument(tracing::debug_span!("request", %id, %method)))
105    }
106}

109pub(crate) struct CatchPanic<S>(pub S);
110
111impl<S> Service<tower_lsp::jsonrpc::Request> for CatchPanic<S>
112where
113    S: Service<
114            tower_lsp::jsonrpc::Request,
115            Response = Option<tower_lsp::jsonrpc::Response>,
116            Error = tower_lsp::ExitedError,
117        >,
118    S::Future: Send + 'static,
119{
120    type Response = S::Response;
121    type Error = CatchPanicError;
122    type Future =
123        Pin<Box<dyn Future<Output = Result<Self::Response, Self::Error>> + Send + 'static>>;
124
125    fn poll_ready(
126        &mut self,
127        cx: &mut std::task::Context<'_>,
128    ) -> std::task::Poll<Result<(), Self::Error>> {
129        self.0.poll_ready(cx).map_err(CatchPanicError::Exited)
130    }
131
132    fn call(&mut self, req: tower_lsp::jsonrpc::Request) -> Self::Future {
133        let fut = AssertUnwindSafe(self.0.call(req)).catch_unwind();
134
135        Box::pin(
136            async move {
137                match fut.await {
138                    Ok(result) => result.map_err(CatchPanicError::Exited),
139                    Err(payload) => Err(CatchPanicError::Panic(PanicMessage::new(payload))),
140                }
141            }
142            .in_current_span(),
143        )
144    }
145}

147/// Errors that may occur when you catch panics in a [`tower_service::Service`].
148#[derive(Debug, Clone)]
149pub enum CatchPanicError {
150    Panic(PanicMessage),
151    Exited(tower_lsp::ExitedError),
152}
153
154impl std::error::Error for CatchPanicError {
155    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
156        match self {
157            CatchPanicError::Panic(p) => Some(p),
158            CatchPanicError::Exited(e) => Some(e),
159        }
160    }
161}
162
163impl std::fmt::Display for CatchPanicError {
164    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
165        match self {
166            CatchPanicError::Panic(p) => write!(f, "{p}"),
167            CatchPanicError::Exited(e) => write!(f, "{e}"),
168        }
169    }
170}

172/// A printable panic payload.
173#[derive(Debug, Clone)]
174pub struct PanicMessage {
175    msg: Option<String>,
176}
177
178impl PanicMessage {
179    fn new(payload: Box<dyn Any + Send>) -> Self {
180        let msg = if let Some(msg) = payload.downcast_ref::<String>() {
181            Some(msg.as_str())
182        } else if let Some(&msg) = payload.downcast_ref::<&str>() {
183            Some(msg)
184        } else {
185            None
186        };
187
188        PanicMessage {
189            msg: msg.map(String::from),
190        }
191    }
192}
193
194impl std::error::Error for PanicMessage {}
195
196impl std::fmt::Display for PanicMessage {
197    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
198        let msg = self.msg.as_deref().unwrap_or("<unknown>");
199        write!(f, "A panic occurred while handling the request: {msg}")
200    }
201}
