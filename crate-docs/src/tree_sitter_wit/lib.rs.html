1//! This crate provides Wit language support for the [tree-sitter][] parsing library.
2//!
3//! Typically, you will use the [language][language func] function to add this language to a
4//! tree-sitter [Parser][], and then use the parser to parse some code:
5//!
6//! ```
7//! let code = r#"
8//! "#;
9//! let mut parser = tree_sitter::Parser::new();
10//! parser.set_language(&tree_sitter_wit::language()).expect("Error loading Wit grammar");
11//! let tree = parser.parse(code, None).unwrap();
12//! assert!(!tree.root_node().has_error());
13//! ```
14//!
15//! [Language]: https://docs.rs/tree-sitter/*/tree_sitter/struct.Language.html
16//! [language func]: fn.language.html
17//! [Parser]: https://docs.rs/tree-sitter/*/tree_sitter/struct.Parser.html
18//! [tree-sitter]: https://tree-sitter.github.io/
19
20use tree_sitter::Language;
21
22extern "C" {
23    fn tree_sitter_wit() -> Language;
24}
25
26/// Get the tree-sitter [Language][] for this grammar.
27///
28/// [Language]: https://docs.rs/tree-sitter/*/tree_sitter/struct.Language.html
29pub fn language() -> Language {
30    unsafe { tree_sitter_wit() }
31}

50#[cfg(test)]
51mod tests {
52    use std::path::Path;
53
54    use super::*;
55
56    #[test]
57    fn test_can_load_grammar() {
58        let mut parser = tree_sitter::Parser::new();
59        parser
60            .set_language(&language())
61            .expect("Error loading Wit grammar");
62    }
63
64    #[test]
65    fn all_queries_are_valid() {
66        let queries_dir = project_root().join("queries");
67        let lang = language();
68
69        for entry in queries_dir.read_dir().unwrap() {
70            let path = entry.unwrap().path();
71
72            if path.extension() != Some("scm".as_ref()) {
73                continue;
74            }
75
76            let query = std::fs::read_to_string(&path).unwrap();
77            let _ = tree_sitter::Query::new(&lang, &query).unwrap();
78        }
79    }
80
81    fn project_root() -> &'static Path {
82        Path::new(env!("CARGO_MANIFEST_DIR"))
83            .ancestors()
84            .find(|p| p.join(".git").exists())
85            .expect("Unable to find the project's root directory")
86    }
