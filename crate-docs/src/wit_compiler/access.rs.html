20macro_rules! node_kinds {
21    ($( $name:ident => ($ast_node:ident, $meta:path)),+ $(,)?) => {
22        paste::paste! {
23            $(
24                #[doc = "A type tag for a [`crate::ast::" $ast_node "`] node."]
25                #[derive(Debug, Copy, Clone, Hash, PartialEq, Eq, PartialOrd, Ord)]
26                pub enum $name {}
27
28                impl NodeKind for $name {
29                    type Ast<'tree> = crate::ast::$ast_node<'tree>;
30                    type Metadata = $meta;
31                }
32
33                #[doc = "A [`Pointer`] for a [`" $name "`]."]
34                pub type [< $name Ptr >] = Pointer<$name>;
35
36                #[doc = "An [`Index`] for a [`" $name "`]."]
37                pub type [< $name Index >] = Index<$name>;
38            )*
39
40            #[derive(Debug, Copy, Clone, Hash, PartialEq, Eq, PartialOrd, Ord)]
41            pub enum AnyIndex {
42                $(
43                    $name(Index<$name>),
44                )*
45            }
46
47            impl AnyIndex {
48                pub fn file(self) -> FilePath {
49                    match self {
50                        $(
51                            AnyIndex::$name(ix) => ix.file(),
52                        )*
53                    }
54                }
55
56                $(
57                    pub fn [< as_ $name:snake >](self) -> Option<Index<$name>> {
58                        match self {
59                            AnyIndex::$name(ix) => Some(ix),
60                            _ => None,
61                        }
62                    }
63                )*
64            }
65
66            $(
67                impl From<Index<$name>> for AnyIndex {
68                    fn from(value: Index<$name>) -> Self {
69                        AnyIndex::$name(value)
70                    }
71                }
72            )*
73        }
74    };
75}

99/// A unique identifier that can be used to refer to an element in a particular
100/// file.
101pub struct Index<K> {
102    file: FilePath,
103    index: RawIndex,
104    _ty: PhantomData<K>,
105}
106
107impl<K: NodeKind> Index<K> {
108    pub(crate) fn new(file: FilePath, index: RawIndex) -> Self {
109        Index {
110            file,
111            index,
112            _ty: PhantomData,
113        }
114    }
115
116    pub fn file(self) -> FilePath {
117        self.file
118    }
119}
120
121impl<K> Ord for Index<K> {
122    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
123        let Index { file, index, _ty } = self;
124        file.cmp(&other.file).then(index.cmp(&other.index))
125    }
126}
127
128impl<K> PartialOrd for Index<K> {
129    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
130        Some(self.cmp(other))
131    }
132}
133
134impl<K> Eq for Index<K> {}
135
136impl<K> PartialEq for Index<K> {
137    fn eq(&self, other: &Self) -> bool {
138        let Index { file, index, _ty } = *self;
139        file == other.file && index == other.index
140    }
141}
142
143impl<K> std::hash::Hash for Index<K> {
144    fn hash<H: std::hash::Hasher>(&self, state: &mut H) {
145        let Index { file, index, _ty } = self;
146        file.hash(state);
147        index.hash(state);
148    }
149}
150
151impl<K> Clone for Index<K> {
152    fn clone(&self) -> Self {
153        *self
154    }
155}
156
157impl<K> Copy for Index<K> {}
158
159impl<K> Debug for Index<K> {
160    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
161        let name = std::any::type_name::<K>().split("::").last().unwrap();
162        let index_name = format!("{name}Index");
163
164        let Index { file, index, _ty } = self;
165
166        f.debug_struct(&index_name)
167            .field("file", file)
168            .field("index", index)
169            .finish()
170    }
171}

191#[derive(Debug, Copy, Clone, Hash, PartialEq, Eq, PartialOrd, Ord)]
192pub enum ScopeIndex {
193    World(WorldIndex),
194    Interface(InterfaceIndex),
195    InlineInterface(InlineInterfaceIndex),
196}
197
198impl ScopeIndex {
199    pub fn file(self) -> FilePath {
200        match self {
201            ScopeIndex::World(w) => w.file(),
202            ScopeIndex::Interface(i) => i.file(),
203            ScopeIndex::InlineInterface(i) => i.file(),
204        }
205    }
206}
207
208impl From<InterfaceIndex> for ScopeIndex {
209    fn from(v: InterfaceIndex) -> Self {
210        Self::Interface(v)
211    }
212}
213
214impl From<InlineInterfaceIndex> for ScopeIndex {
215    fn from(v: InlineInterfaceIndex) -> Self {
216        Self::InlineInterface(v)
217    }
218}
219
220impl From<WorldIndex> for ScopeIndex {
221    fn from(v: WorldIndex) -> Self {
222        Self::World(v)
223    }
224}
225
226impl From<ScopeIndex> for AnyIndex {
227    fn from(value: ScopeIndex) -> Self {
228        match value {
229            ScopeIndex::World(w) => w.into(),
230            ScopeIndex::Interface(i) => i.into(),
231            ScopeIndex::InlineInterface(i) => i.into(),
232        }
233    }
234}

236/// An index optimised for use in item IDs.
237///
238/// You typically won't use this directly, and instead rely on strongly-typed
239/// wrappers.
240///
241/// # Implementation
242///
243/// Under the hood, the index is represented as a [`NonZeroU16`].  We make the
244/// assumption that no file will contain more than `2^16-2` sequential elements
245/// of the same type, so we can get away with only using 2 bytes for our indices
246/// rather than the 8 we would need if we stored a `usize`.
247///
248/// Strongly typed wrappers will sometimes include enums, so by using
249/// [`NonZeroU16`] over [`u16`], we are more likely to benefit from niche
250/// optimisations.
251#[derive(Debug, Copy, Clone, Hash, PartialEq, Eq, PartialOrd, Ord)]
252#[repr(transparent)]
253pub(crate) struct RawIndex(NonZeroU16);
254
255impl RawIndex {
256    const MAX: u16 = u16::MAX - 1;
257    pub const ZERO: RawIndex = RawIndex::new(0);
258
259    pub(crate) const fn new(raw: usize) -> Self {
260        assert!(raw <= RawIndex::MAX as usize);
261
262        let raw = match (raw as u16).checked_add(1) {
263            Some(raw) => raw,
264            None => panic!(),
265        };
266
267        match NonZeroU16::new(raw) {
268            Some(raw) => RawIndex(raw),
269            None => panic!(),
270        }
271    }
272
273    pub(crate) fn next(self) -> Self {
274        RawIndex::new(self.as_usize().checked_add(1).unwrap())
275    }
276
277    pub const fn as_usize(self) -> usize {
278        self.0.get() as usize - 1
279    }
280}

282pub struct Pointer<K> {
283    location: Location,
284    _ty: PhantomData<K>,
285}
286
287impl<K: NodeKind> Pointer<K> {
288    pub fn for_node(file: FilePath, node: K::Ast<'_>) -> Self {
289        let range = node.range();
290        let location = Location::new(file, range);
291
292        Pointer {
293            location,
294            _ty: PhantomData,
295        }
296    }
297
298    /// Get the [`crate::ast`] node from the AST [`Tree`].
299    pub fn ast_node(self, tree: &Tree) -> K::Ast<'_> {
300        tree.find(self.location.range)
301    }
302}
303
304impl<K> Pointer<K> {
305    pub fn file(self) -> FilePath {
306        self.location.filename
307    }
308
309    pub fn range(self) -> tree_sitter::Range {
310        self.location.range
311    }
312
313    pub fn location(self) -> crate::diagnostics::Location {
314        self.location
315    }
316}
