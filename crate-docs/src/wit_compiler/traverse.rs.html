10/// Trait which represents a stateful cursor in a n-ary tree.
11/// The cursor can be moved between nodes in the tree by the given methods,
12/// and the node which the cursor is currently pointing at can be read as well.
13pub trait Cursor {
14    /// The type of the nodes which the cursor points at; the cursor is always pointing
15    /// at exactly one of this type.
16    type Node;
17
18    /// Move this cursor to the first child of its current node.
19    ///
20    /// This returns `true` if the cursor successfully moved, and returns `false`
21    /// if there were no children.
22    fn goto_first_child(&mut self) -> bool;
23
24    /// Move this cursor to the parent of its current node.
25    ///
26    /// This returns `true` if the cursor successfully moved, and returns `false`
27    /// if there was no parent node (the cursor was already on the root node).
28    fn goto_parent(&mut self) -> bool;
29
30    /// Move this cursor to the next sibling of its current node.
31    ///
32    /// This returns `true` if the cursor successfully moved, and returns `false`
33    /// if there was no next sibling node.
34    fn goto_next_sibling(&mut self) -> bool;
35
36    /// Get the node which the cursor is currently pointing at.
37    fn node(&self) -> Self::Node;
38}
39
40impl<T> Cursor for &mut T
41where
42    T: Cursor,
43{
44    type Node = T::Node;
45
46    fn goto_first_child(&mut self) -> bool {
47        T::goto_first_child(self)
48    }
49
50    fn goto_parent(&mut self) -> bool {
51        T::goto_parent(self)
52    }
53
54    fn goto_next_sibling(&mut self) -> bool {
55        T::goto_next_sibling(self)
56    }
57
58    fn node(&self) -> Self::Node {
59        T::node(self)
60    }
61}

86/// Order to iterate through a n-ary tree; for n-ary trees only
87/// Pre-order and Post-order make sense.
88#[derive(Eq, PartialEq, Hash, Debug, Copy, Clone)]
89pub enum Order {
90    Pre,
91    Post,
92}
93
94/// Iterative traversal of the tree; serves as a reference for both
95/// PreorderTraversal and PostorderTraversal, as they both will call the exact same
96/// cursor methods in the exact same order as this function for a given tree; the order
97/// is also the same as traverse_recursive.
98#[allow(dead_code)]
99fn traverse_iterative<C: Cursor, F>(mut c: C, order: Order, mut cb: F)
100where
101    F: FnMut(C::Node),
102{
103    loop {
104        // This is the first time we've encountered the node, so we'll call if preorder
105        if order == Order::Pre {
106            cb(c.node());
107        }
108
109        // Keep travelling down the tree as far as we can
110        if c.goto_first_child() {
111            continue;
112        }
113
114        let node = c.node();
115
116        // If we can't travel any further down, try going to next sibling and repeating
117        if c.goto_next_sibling() {
118            // If we succeed in going to the previous nodes sibling,
119            // we won't be encountering that node again, so we'll call if postorder
120            if order == Order::Post {
121                cb(node);
122            }
123            continue;
124        }
125
126        // Otherwise, we must travel back up; we'll loop until we reach the root or can
127        // go to the next sibling of a node again.
128        loop {
129            // Since we're retracing back up the tree, this is the last time we'll encounter
130            // this node, so we'll call if postorder
131            if order == Order::Post {
132                cb(c.node());
133            }
134            if !c.goto_parent() {
135                // We have arrived back at the root, so we are done.
136                return;
137            }
138
139            let node = c.node();
140
141            if c.goto_next_sibling() {
142                // If we succeed in going to the previous node's sibling,
143                // we will go back to travelling down that sibling's tree, and we also
144                // won't be encountering the previous node again, so we'll call if postorder
145                if order == Order::Post {
146                    cb(node);
147                }
148                break;
149            }
150        }
151    }
152}

154/// Idiomatic recursive traversal of the tree; this version is easier to understand
155/// conceptually, but the recursion is actually unnecessary and can cause stack overflow.
156#[allow(dead_code)]
157fn traverse_recursive<C: Cursor, F>(mut c: C, order: Order, mut cb: F)
158where
159    F: FnMut(C::Node),
160{
161    traverse_helper(&mut c, order, &mut cb);
162}
163
164fn traverse_helper<C: Cursor, F>(c: &mut C, order: Order, cb: &mut F)
165where
166    F: FnMut(C::Node),
167{
168    // If preorder, call the callback when we first touch the node
169    if order == Order::Pre {
170        cb(c.node());
171    }
172    if c.goto_first_child() {
173        // If there is a child, recursively call on
174        // that child and all its siblings
175        loop {
176            traverse_helper(c, order, cb);
177            if !c.goto_next_sibling() {
178                break;
179            }
180        }
181        // Make sure to reset back to the original node;
182        // this must always return true, as we only get here if we go to a child
183        // of the original node.
184        assert!(c.goto_parent());
185    }
186    // If preorder, call the callback after the recursive calls on child nodes
187    if order == Order::Post {
188        cb(c.node());
189    }
190}

192struct PreorderTraverse<C> {
193    cursor: Option<C>,
194}
195
196impl<C> PreorderTraverse<C> {
197    pub fn new(c: C) -> Self {
198        PreorderTraverse { cursor: Some(c) }
199    }
200}
201
202impl<C> Iterator for PreorderTraverse<C>
203where
204    C: Cursor,
205{
206    type Item = C::Node;
207
208    fn next(&mut self) -> Option<Self::Item> {
209        let c = match self.cursor.as_mut() {
210            None => {
211                return None;
212            }
213            Some(c) => c,
214        };
215
216        // We will always return the node we were on at the start;
217        // the node we traverse to will either be returned on the next iteration,
218        // or will be back to the root node, at which point we'll clear out
219        // the reference to the cursor
220        let node = c.node();
221
222        // First, try to go to a child or a sibling; if either succeed, this will be the
223        // first time we touch that node, so it'll be the next starting node
224        if c.goto_first_child() || c.goto_next_sibling() {
225            return Some(node);
226        }
227
228        loop {
229            // If we can't go to the parent, then that means we've reached the root, and our
230            // iterator will be done in the next iteration
231            if !c.goto_parent() {
232                self.cursor = None;
233                break;
234            }
235
236            // If we get to a sibling, then this will be the first time we touch that node,
237            // so it'll be the next starting node
238            if c.goto_next_sibling() {
239                break;
240            }
241        }
242
243        Some(node)
244    }
245}

247struct PostorderTraverse<C> {
248    cursor: Option<C>,
249    retracing: bool,
250}
251
252impl<C> PostorderTraverse<C> {
253    pub fn new(c: C) -> Self {
254        PostorderTraverse {
255            cursor: Some(c),
256            retracing: false,
257        }
258    }
259}
260
261impl<C> Iterator for PostorderTraverse<C>
262where
263    C: Cursor,
264{
265    type Item = C::Node;
266
267    fn next(&mut self) -> Option<Self::Item> {
268        let c = match self.cursor.as_mut() {
269            None => {
270                return None;
271            }
272            Some(c) => c,
273        };
274
275        // For the postorder traversal, we will only return a node when we are travelling back up
276        // the tree structure. Therefore, we go all the way to the leaves of the tree immediately,
277        // and only when we are retracing do we return elements
278        if !self.retracing {
279            while c.goto_first_child() {}
280        }
281
282        // Much like in preorder traversal, we want to return the node we were previously at.
283        // We know this will be the last time we touch this node, as we will either be going
284        // to its next sibling or retracing back up the tree
285        let node = c.node();
286        if c.goto_next_sibling() {
287            // If we successfully go to a sibling of this node, we want to go back down
288            // the tree on the next iteration
289            self.retracing = false;
290        } else {
291            // If we weren't already retracing, we are now; travel upwards until we can
292            // go to the next sibling or reach the root again
293            self.retracing = true;
294            if !c.goto_parent() {
295                // We've reached the root again, and our iteration is done
296                self.cursor = None;
297            }
298        }
299
300        Some(node)
301    }
302}

304enum Traverse<C> {
305    Post(PostorderTraverse<C>),
306    Pre(PreorderTraverse<C>),
307}
308
309impl<C> Traverse<C> {
310    pub fn new(c: C, order: Order) -> Self {
311        match order {
312            Order::Pre => Traverse::Pre(PreorderTraverse::new(c)),
313            Order::Post => Traverse::Post(PostorderTraverse::new(c)),
314        }
315    }
316}

318impl<'a> Traverse<tree_sitter::TreeCursor<'a>> {
319    #[allow(dead_code)]
320    pub fn from_tree(tree: &'a tree_sitter::Tree, order: Order) -> Self {
321        Traverse::new(tree.walk(), order)
322    }
323}

325/// Convenience method to traverse a tree-sitter [`Tree`] in an order according to `order`.
326///
327/// [`Tree`]: tree_sitter::Tree
328pub fn tree(
329    tree: &tree_sitter::Tree,
330    order: Order,
331) -> impl FusedIterator<Item = tree_sitter::Node<'_>> {
332    cursor(tree.walk(), order)
333}

335/// Traverse an n-ary tree using `cursor`, returning the nodes of the tree
336/// through an iterator in an order according to `order`.
337pub fn cursor<C: Cursor>(cursor: C, order: Order) -> impl FusedIterator<Item = C::Node> {
338    Traverse::new(cursor, order)
339}

341impl<C> Iterator for Traverse<C>
342where
343    C: Cursor,
344{
345    type Item = C::Node;
346
347    fn next(&mut self) -> Option<Self::Item> {
348        match self {
349            Traverse::Post(i) => i.next(),
350            Traverse::Pre(i) => i.next(),
351        }
352    }
353}
