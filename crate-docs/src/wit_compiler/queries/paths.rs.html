13/// Resolve an interface or world path, including file-local aliases.
14/// Alias traversal stays within the referring file and terminates on cycles.
15pub fn resolve_use_path<'db>(
16    db: &'db dyn Db,
17    ws: Workspace,
18    file: SourceFile,
19    path: ast::UsePath<'db>,
20) -> Option<TopLevelItemMetadata> {
21    resolve_path(db, ws, file, path).map(|(target, _)| target)
22}

34fn resolve_path<'db>(
35    db: &'db dyn Db,
36    ws: Workspace,
37    file: SourceFile,
38    mut path: ast::UsePath<'db>,
39) -> Option<(TopLevelItemMetadata, Location)> {
40    let mut binding = None;
41    let packages = super::workspace_packages(db, ws);
42    let own_package = packages
43        .iter()
44        .copied()
45        .find(|p| p.contains(db, file.path(db)))?;
46    let ast = super::parse(db, file);
47    let src = file.contents(db);
48    let mut visited = HashSet::new();
49
50    loop {
51        if !visited.insert(path.range().start_byte) {
52            return None;
53        }
54        if let Some(id) = path.id() {
55            let name = Ident::from_ast(db, id, src);
56            let alias = ast
57                .source_file(db)
58                .iter_top_level_items()
59                .filter_map(|item| item.toplevel_use_item())
60                .find_map(|item| {
61                    let target = item.path()?;
62                    let local = item.iter_alias().next().or_else(|| path_name(target))?;
63                    (Ident::from_ast(db, local, src) == name)
64                        .then_some((target, item.iter_alias().next()))
65                });
66            if let Some((target, alias)) = alias {
67                if binding.is_none() {
68                    binding = alias.map(|id| Location::new(file.path(db), id.range()));
69                }
70                path = target;
71                continue;
72            }
73            return super::package_items(db, own_package)
74                .items_by_name(db)
75                .get(&name)
76                .copied()
77                .map(|target| (target, binding.unwrap_or_else(|| target.definition(db))));
78        }
79
80        let qualified = path.fully_qualified_use_path()?;
81        let mut ids = qualified.iter_ids();
82        let namespace = Ident::from_ast(db, ids.next()?, src);
83        let package = Ident::from_ast(db, ids.next()?, src);
84        let item = Ident::from_ast(db, ids.next()?, src);
85        let mut package_id = format!("{}:{}", namespace.raw(db), package.raw(db));
86        if let Some(version) = qualified.iter_valid_semvers().next() {
87            package_id.push('@');
88            package_id.push_str(version.utf8_text(src));
89        }
90        let package = packages
91            .iter()
92            .copied()
93            .find(|p| p.id(db).is_some_and(|id| id.raw(db).as_str() == package_id))?;
94        return super::package_items(db, package)
95            .items_by_name(db)
96            .get(&item)
97            .copied()
98            .map(|target| (target, binding.unwrap_or_else(|| target.definition(db))));
99    }
100}

102fn path_name(path: ast::UsePath<'_>) -> Option<ast::Id<'_>> {
103    path.id()
104        .or_else(|| path.fully_qualified_use_path()?.iter_ids().last())
