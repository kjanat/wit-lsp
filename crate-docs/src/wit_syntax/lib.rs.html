20pub(crate) fn children(parent: Node<'_>) -> impl Iterator<Item = Node<'_>> {
21    std::iter::successors(parent.child(0), |last| last.next_sibling())

24/// Get the underlying name an identifier is referring to. This removes
25/// %-escaping.
26pub(crate) fn ident(raw: &str) -> &str {
27    raw.strip_prefix('%').unwrap_or(raw)

30#[cfg(test)]
31mod tests {
32    use super::*;
33
34    #[test]
35    fn field_iterator_outlives_node_wrapper() {
36        let src = "/// Interface documentation\ninterface example {}";
37        let mut parser = tree_sitter::Parser::new();
38        parser.set_language(&tree_sitter_wit::language()).unwrap();
39        let tree = parser.parse(src, None).unwrap();
40        assert!(!tree.root_node().has_error());
41
42        let mut attributes = {
43            let interface = SourceFile::cast(tree.root_node())
44                .unwrap()
45                .iter_top_level_items()
46                .find_map(TopLevelItem::interface_item)
47                .unwrap();
48            let borrowed = &interface;
49            borrowed.iter_attributes()
50        };
51
52        assert_eq!(
53            attributes.next().unwrap().utf8_text(src).trim(),
54            "/// Interface documentation"
55        );
56        assert!(attributes.next().is_none());
57    }
58}
