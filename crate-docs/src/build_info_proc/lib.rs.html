14/// Call as `build_info!(fn name)` to create a function called `name` that returns a reference to a lazily created and
15/// cached `BuildInfo` object.
16///
17/// This macro also accepts a visibility specifier for the generated function, such as `build_info!(pub fn version)`.
18#[cfg(feature = "runtime")]
19#[manyhow::manyhow]
20#[proc_macro]
21pub fn build_info(input: TokenStream) -> manyhow::Result<TokenStream2> {
22	function::build_info(input, deserialize_build_info()?)
23}
24
25#[manyhow::manyhow]
26#[proc_macro]
27pub fn format(input: TokenStream, emitter: &mut Emitter) -> manyhow::Result<TokenStream2> {
28	let result = format::format(input, deserialize_build_info()?, emitter)?;
29	emitter.into_result()?;
30	Ok(result)
31}

33fn deserialize_build_info() -> manyhow::Result<BuildInfo> {
34	let data = std::env::var("BUILD_INFO").map_err(|err| {
35		manyhow::error_message!("No BuildInfo data found!";
36			note = "Did you call build_info_build::build_script() in your build.rs?";
37			note = "This crate expects version {} of the BuildInfo data", build_info_common::crate_version();
38			note = "Caused by: {}", err
39		)
40	})?;
41
42	// println!("Serialized data is {} bytes long.", data.len());
43
44	let versioned: VersionedString = serde_json::from_str(&data).map_err(|err| {
45		manyhow::error_message!("Could not deserialize BuildInfo data at all!";
46			note = "This crate expects version {} of the BuildInfo data", build_info_common::crate_version();
47			note = "Caused by: {}", err;
48		)
49	})?;
50
51	if !versioned.check() {
52		// TODO: This should really be a warning - but warnings are currently nightly-only...
53		manyhow::bail!("BuildInfo data has an different version!";
54			note = "The serialized data has version {}", versioned.version;
55			note = "This crate expects version {} of the BuildInfo data", build_info_common::crate_version();
56		);
57	}
58
59	let bytes = z85::decode(versioned.string.as_bytes()).map_err(|err| {
60		manyhow::error_message!("BuildInfo data cannot be deserialized!";
61			note = "The serialized data has version {}", versioned.version;
62			note = "This crate expects version {} of the BuildInfo data", build_info_common::crate_version();
63			note = "Underlying cause: {}", err;
64		)
65	})?;
66	let cursor = Cursor::new(&bytes);
67	let mut decoder = zstd::Decoder::new(cursor).expect("Could not crate ZSTD decoder");
68	ciborium::from_reader(&mut decoder).map_err(|err| {
69		manyhow::error_message!("BuildInfo data cannot be deserialized!";
70			note = "The serialized data has version {}", versioned.version;
71			note = "This crate expects version {} of the BuildInfo data", build_info_common::crate_version();
72			note = "Underlying cause: {}", err;
73		)
74		.into()
75	})
76}
