1use std::any::Any;

3use build_info_common::BuildInfo;

5use super::{FormatSpecifier, OP_FIELD_ACCESS, Type, Value, as_arguments_0, as_field_name};

7impl Value for BuildInfo {
8	fn call(&self, func: &str, args: &[Box<dyn Value>]) -> manyhow::Result<Box<dyn Value>> {
9		match func {
10			OP_FIELD_ACCESS => match as_field_name(args) {
11				"timestamp" => Ok(Box::new(self.timestamp)),
12				"profile" => Ok(Box::new(self.profile.clone())),
13				"optimization_level" => Ok(Box::new(self.optimization_level)),
14				"crate_info" => Ok(Box::new(self.crate_info.clone())),
15				"target" => Ok(Box::new(self.target.clone())),
16				"compiler" => Ok(Box::new(self.compiler.clone())),
17				"version_control" => Ok(Box::new(self.version_control.clone())),
18				_ => self.call_base(func, args),
19			},
20			"to_string" => {
21				as_arguments_0(args)?;
22				Ok(Box::new(self.to_string()))
23			}
24			_ => self.call_base(func, args),
25		}
26	}
27
28	fn get_type(&self) -> Type {
29		Type::BuildInfo
30	}
31
32	fn as_any(&self) -> &dyn Any {
33		self
34	}
35
36	fn format(&self, buffer: &mut String, spec: FormatSpecifier) {
37		use std::fmt::Write;
38
39		match spec {
40			FormatSpecifier::Default => write!(buffer, "{self}").unwrap(),
41			FormatSpecifier::Debug => write!(buffer, "{self:?}").unwrap(),
42			FormatSpecifier::DebugAlt => write!(buffer, "{self:#?}").unwrap(),
43		}
44	}
45}
