5/// "Options" are flags that can be supplied to the various salsa related
6/// macros. They are listed like `(ref, no_eq, foo=bar)` etc. The commas
7/// are required and trailing commas are permitted. The options accepted
8/// for any particular location are configured via the `AllowedOptions`
9/// trait.
10pub(crate) struct Options<A: AllowedOptions> {
11    /// The `return_ref` option is used to signal that field/return type is "by ref"
12    ///
13    /// If this is `Some`, the value is the `ref` identifier.
14    pub return_ref: Option<syn::Ident>,
15
16    ///  The `no_eq` option is used to signal that a given field does not implement
17    /// the `Eq` trait and cannot be compared for equality.
18    ///
19    /// If this is `Some`, the value is the `no_eq` identifier.
20    pub no_eq: Option<syn::Ident>,
21
22    /// The `singleton` option is used on input with only one field
23    /// It allows the creation of convenient methods
24    pub singleton: Option<syn::Ident>,
25
26    /// The `specify` option is used to signal that a tracked function can
27    /// have its value externally specified (at least some of the time).
28    ///
29    /// If this is `Some`, the value is the `specify` identifier.
30    pub specify: Option<syn::Ident>,
31
32    /// The `jar = <type>` option is used to indicate the jar; it defaults to `crate::jar`.
33    ///
34    /// If this is `Some`, the value is the `<type>`.
35    pub jar_ty: Option<syn::Type>,
36
37    /// The `db = <path>` option is used to indicate the db.
38    ///
39    /// If this is `Some`, the value is the `<path>`.
40    pub db_path: Option<syn::Path>,
41
42    /// The `recovery_fn = <path>` option is used to indicate the recovery function.
43    ///
44    /// If this is `Some`, the value is the `<path>`.
45    pub recovery_fn: Option<syn::Path>,
46
47    /// The `data = <ident>` option is used to define the name of the data type for an interned
48    /// struct.
49    ///
50    /// If this is `Some`, the value is the `<ident>`.
51    pub data: Option<syn::Ident>,
52
53    /// The `lru = <usize>` option is used to set the lru capacity for a tracked function.
54    ///
55    /// If this is `Some`, the value is the `<usize>`.
56    pub lru: Option<usize>,
57
58    /// The `constructor = <ident>` option lets the user specify the name of
59    /// the constructor of a salsa struct.
60    ///
61    /// If this is `Some`, the value is the `<ident>`.
62    pub constructor_name: Option<syn::Ident>,
63
64    /// Remember the `A` parameter, which plays no role after parsing.
65    phantom: PhantomData<A>,
66}
67
68impl<A: AllowedOptions> Default for Options<A> {
69    fn default() -> Self {
70        Self {
71            return_ref: Default::default(),
72            specify: Default::default(),
73            no_eq: Default::default(),
74            jar_ty: Default::default(),
75            db_path: Default::default(),
76            recovery_fn: Default::default(),
77            data: Default::default(),
78            constructor_name: Default::default(),
79            phantom: Default::default(),
80            lru: Default::default(),
81            singleton: Default::default(),
82        }
83    }
84}

86/// These flags determine which options are allowed in a given context
87pub(crate) trait AllowedOptions {
88    const RETURN_REF: bool;
89    const SPECIFY: bool;
90    const NO_EQ: bool;
91    const SINGLETON: bool;
92    const JAR: bool;
93    const DATA: bool;
94    const DB: bool;
95    const RECOVERY_FN: bool;
96    const LRU: bool;
97    const CONSTRUCTOR_NAME: bool;
98}
99
100type Equals = syn::Token![=];
101type Comma = syn::Token![,];
102
103impl<A: AllowedOptions> Options<A> {
104    /// Returns the `jar type` given by the user; if none is given,
105    /// returns the default `crate::Jar`.
106    pub(crate) fn jar_ty(&self) -> syn::Type {
107        if let Some(jar_ty) = &self.jar_ty {
108            return jar_ty.clone();
109        }
110
111        parse_quote! {crate::Jar}
112    }
113
114    pub(crate) fn should_backdate(&self) -> bool {
115        self.no_eq.is_none()
116    }
117}

119impl<A: AllowedOptions> syn::parse::Parse for Options<A> {
120    fn parse(input: syn::parse::ParseStream) -> syn::Result<Self> {
121        let mut options = Options::default();
122
123        while !input.is_empty() {
124            let ident: syn::Ident = syn::Ident::parse_any(input)?;
125            if ident == "return_ref" {
126                if A::RETURN_REF {
127                    if let Some(old) = std::mem::replace(&mut options.return_ref, Some(ident)) {
128                        return Err(syn::Error::new(
129                            old.span(),
130                            "option `return_ref` provided twice",
131                        ));
132                    }
133                } else {
134                    return Err(syn::Error::new(
135                        ident.span(),
136                        "`return_ref` option not allowed here",
137                    ));
138                }
139            } else if ident == "no_eq" {
140                if A::NO_EQ {
141                    if let Some(old) = std::mem::replace(&mut options.no_eq, Some(ident)) {
142                        return Err(syn::Error::new(old.span(), "option `no_eq` provided twice"));
143                    }
144                } else {
145                    return Err(syn::Error::new(
146                        ident.span(),
147                        "`no_eq` option not allowed here",
148                    ));
149                }
150            } else if ident == "singleton" {
151                if A::SINGLETON {
152                    if let Some(old) = std::mem::replace(&mut options.singleton, Some(ident)) {
153                        return Err(syn::Error::new(
154                            old.span(),
155                            "option `singleton` provided twice",
156                        ));
157                    }
158                } else {
159                    return Err(syn::Error::new(
160                        ident.span(),
161                        "`singleton` option not allowed here",
162                    ));
163                }
164            } else if ident == "specify" {
165                if A::SPECIFY {
166                    if let Some(old) = std::mem::replace(&mut options.specify, Some(ident)) {
167                        return Err(syn::Error::new(
168                            old.span(),
169                            "option `specify` provided twice",
170                        ));
171                    }
172                } else {
173                    return Err(syn::Error::new(
174                        ident.span(),
175                        "`specify` option not allowed here",
176                    ));
177                }
178            } else if ident == "jar" {
179                if A::JAR {
180                    let _eq = Equals::parse(input)?;
181                    let ty = syn::Type::parse(input)?;
182                    if let Some(old) = std::mem::replace(&mut options.jar_ty, Some(ty)) {
183                        return Err(syn::Error::new(old.span(), "option `jar` provided twice"));
184                    }
185                } else {
186                    return Err(syn::Error::new(
187                        ident.span(),
188                        "`jar` option not allowed here",
189                    ));
190                }
191            } else if ident == "db" {
192                if A::DB {
193                    let _eq = Equals::parse(input)?;
194                    let path = syn::Path::parse(input)?;
195                    if let Some(old) = std::mem::replace(&mut options.db_path, Some(path)) {
196                        return Err(syn::Error::new(old.span(), "option `db` provided twice"));
197                    }
198                } else {
199                    return Err(syn::Error::new(
200                        ident.span(),
201                        "`db` option not allowed here",
202                    ));
203                }
204            } else if ident == "recovery_fn" {
205                if A::RECOVERY_FN {
206                    let _eq = Equals::parse(input)?;
207                    let path = syn::Path::parse(input)?;
208                    if let Some(old) = std::mem::replace(&mut options.recovery_fn, Some(path)) {
209                        return Err(syn::Error::new(
210                            old.span(),
211                            "option `recovery_fn` provided twice",
212                        ));
213                    }
214                } else {
215                    return Err(syn::Error::new(
216                        ident.span(),
217                        "`recovery_fn` option not allowed here",
218                    ));
219                }
220            } else if ident == "data" {
221                if A::DATA {
222                    let _eq = Equals::parse(input)?;
223                    let ident = syn::Ident::parse(input)?;
224                    if let Some(old) = std::mem::replace(&mut options.data, Some(ident)) {
225                        return Err(syn::Error::new(old.span(), "option `data` provided twice"));
226                    }
227                } else {
228                    return Err(syn::Error::new(
229                        ident.span(),
230                        "`data` option not allowed here",
231                    ));
232                }
233            } else if ident == "lru" {
234                if A::LRU {
235                    let _eq = Equals::parse(input)?;
236                    let lit = syn::LitInt::parse(input)?;
237                    let value = lit.base10_parse::<usize>()?;
238                    if let Some(old) = std::mem::replace(&mut options.lru, Some(value)) {
239                        return Err(syn::Error::new(old.span(), "option `lru` provided twice"));
240                    }
241                } else {
242                    return Err(syn::Error::new(
243                        ident.span(),
244                        "`lru` option not allowed here",
245                    ));
246                }
247            } else if ident == "constructor" {
248                if A::CONSTRUCTOR_NAME {
249                    let _eq = Equals::parse(input)?;
250                    let ident = syn::Ident::parse(input)?;
251                    if let Some(old) = std::mem::replace(&mut options.constructor_name, Some(ident))
252                    {
253                        return Err(syn::Error::new(
254                            old.span(),
255                            "option `constructor` provided twice",
256                        ));
257                    }
258                } else {
259                    return Err(syn::Error::new(
260                        ident.span(),
261                        "`constructor` option not allowed here",
262                    ));
263                }
264            } else {
265                return Err(syn::Error::new(
266                    ident.span(),
267                    format!("unrecognized option `{}`", ident),
268                ));
269            }
270
271            if input.is_empty() {
272                break;
273            }
274
275            let _comma = Comma::parse(input)?;
276        }
277
278        Ok(options)
279    }
