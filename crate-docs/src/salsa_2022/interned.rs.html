26/// The interned ingredient has the job of hashing values of type `Data` to produce an `Id`.
27/// It used to store interned structs but also to store the id fields of a tracked struct.
28/// Interned values endure until they are explicitly removed in some way.
29pub struct InternedIngredient<Id: InternedId, Data: InternedData> {
30    /// Index of this ingredient in the database (used to construct database-ids, etc).
31    ingredient_index: IngredientIndex,
32
33    /// Maps from data to the existing interned id for that data.
34    ///
35    /// Deadlock requirement: We access `value_map` while holding lock on `key_map`, but not vice versa.
36    key_map: FxDashMap<Data, Id>,
37
38    /// Maps from an interned id to its data.
39    ///
40    /// Deadlock requirement: We access `value_map` while holding lock on `key_map`, but not vice versa.
41    value_map: FxDashMap<Id, Box<Data>>,
42
43    /// counter for the next id.
44    counter: AtomicCell<u32>,
45
46    /// Stores the revision when this interned ingredient was last cleared.
47    /// You can clear an interned table at any point, deleting all its entries,
48    /// but that will make anything dependent on those entries dirty and in need
49    /// of being recomputed.
50    reset_at: Revision,
51
52    /// When specific entries are deleted from the interned table, their data is added
53    /// to this vector rather than being immediately freed. This is because we may` have
54    /// references to that data floating about that are tied to the lifetime of some
55    /// `&db` reference. This queue itself is not freed until we have an `&mut db` reference,
56    /// guaranteeing that there are no more references to it.
57    deleted_entries: SegQueue<Box<Data>>,
58
59    debug_name: &'static str,
60}

62impl<Id, Data> InternedIngredient<Id, Data>
63where
64    Id: InternedId,
65    Data: InternedData,
66{
67    pub fn new(ingredient_index: IngredientIndex, debug_name: &'static str) -> Self {
68        Self {
69            ingredient_index,
70            key_map: Default::default(),
71            value_map: Default::default(),
72            counter: AtomicCell::default(),
73            reset_at: Revision::start(),
74            deleted_entries: Default::default(),
75            debug_name,
76        }
77    }
78
79    pub fn intern(&self, runtime: &Runtime, data: Data) -> Id {
80        runtime.report_tracked_read(
81            DependencyIndex::for_table(self.ingredient_index),
82            Durability::MAX,
83            self.reset_at,
84        );
85
86        // Optimisation to only get read lock on the map if the data has already
87        // been interned.
88        if let Some(id) = self.key_map.get(&data) {
89            return *id;
90        }
91
92        match self.key_map.entry(data.clone()) {
93            // Data has been interned by a racing call, use that ID instead
94            dashmap::mapref::entry::Entry::Occupied(entry) => *entry.get(),
95            // We won any races so should intern the data
96            dashmap::mapref::entry::Entry::Vacant(entry) => {
97                let next_id = self.counter.fetch_add(1);
98                let next_id = Id::from_id(crate::id::Id::from_u32(next_id));
99                let old_value = self.value_map.insert(next_id, Box::new(data));
100                assert!(
101                    old_value.is_none(),
102                    "next_id is guaranteed to be unique, bar overflow"
103                );
104                entry.insert(next_id);
105                next_id
106            }
107        }
108    }
109
110    pub(crate) fn reset_at(&self) -> Revision {
111        self.reset_at
112    }
113
114    pub fn reset(&mut self, revision: Revision) {
115        assert!(revision > self.reset_at);
116        self.reset_at = revision;
117        self.key_map.clear();
118        self.value_map.clear();
119    }
120
121    #[track_caller]
122    pub fn data<'db>(&'db self, runtime: &'db Runtime, id: Id) -> &'db Data {
123        runtime.report_tracked_read(
124            DependencyIndex::for_table(self.ingredient_index),
125            Durability::MAX,
126            self.reset_at,
127        );
128
129        let data = match self.value_map.get(&id) {
130            Some(d) => d,
131            None => {
132                panic!("no data found for id `{:?}`", id)
133            }
134        };
135
136        // Unsafety clause:
137        //
138        // * Values are only removed or altered when we have `&mut self`
139        unsafe { transmute_lifetime(self, &**data) }
140    }
141
142    /// Get the ingredient index for this table.
143    pub(super) fn ingredient_index(&self) -> IngredientIndex {
144        self.ingredient_index
145    }
146
147    /// Deletes an index from the interning table, making it available for re-use.
148    ///
149    /// # Warning
150    ///
151    /// This should only be used when you are certain that:
152    ///  1. The given `id` has not (and will not) be used in the current revision.
153    ///  2. The interned data corresponding to `id` will not be interned in this revision.
154    ///
155    /// More specifically, this is used when a query `Q` executes and we can compare the
156    /// entities `E_now` that it produced in this revision vs the entities `E_prev` it
157    /// produced in the last revision. Any missing entities `E_prev - E_new` can be deleted.
158    ///
159    /// If you are wrong about this, it should not be unsafe, but unpredictable results may occur.
160    pub(crate) fn delete_index(&self, id: Id) {
161        let (_, key) = self
162            .value_map
163            .remove(&id)
164            .unwrap_or_else(|| panic!("No entry for id `{:?}`", id));
165
166        self.key_map.remove(&key);
167        // Careful: even though `id` ought not to have been used in this revision,
168        // we don't know that for sure since users could have leaked things. If they did,
169        // they may have stray references into `data`. So push the box onto the
170        // "to be deleted" queue.
171        //
172        // To avoid this, we could include some kind of atomic counter in the `Box` that
173        // gets set whenever `data` executes, so we can track if the data was accessed since
174        // the last time an `&mut self` method was called. But that'd take extra storage
175        // and doesn't obviously seem worth it.
176        self.deleted_entries.push(key);
177    }
178
179    pub(crate) fn clear_deleted_indices(&mut self) {
180        std::mem::take(&mut self.deleted_entries);
181    }
182}

184// Returns `u` but with the lifetime of `t`.
185//
186// Safe if you know that data at `u` will remain shared
187// until the reference `t` expires.
188unsafe fn transmute_lifetime<'t, 'u, T, U>(_t: &'t T, u: &'u U) -> &'t U {
189    std::mem::transmute(u)
190}

192impl<DB: ?Sized, Id, Data> Ingredient<DB> for InternedIngredient<Id, Data>
193where

198        self.ingredient_index
199    }

201    fn maybe_changed_after(&self, _db: &DB, _input: DependencyIndex, revision: Revision) -> bool {
202        revision < self.reset_at
203    }
204
205    fn cycle_recovery_strategy(&self) -> crate::cycle::CycleRecoveryStrategy {
206        crate::cycle::CycleRecoveryStrategy::Panic
207    }

210        None
211    }

213    fn mark_validated_output(
214        &self,
215        _db: &DB,
216        executor: DatabaseKeyIndex,
217        output_key: Option<crate::Id>,
218    ) {
219        unreachable!(
220            "mark_validated_output({:?}, {:?}): input cannot be the output of a tracked function",
221            executor, output_key
222        );
223    }
224
225    fn remove_stale_output(
226        &self,
227        _db: &DB,
228        executor: DatabaseKeyIndex,
229        stale_output_key: Option<crate::Id>,
230    ) {
231        unreachable!(
232            "remove_stale_output({:?}, {:?}): interned ids are not outputs",
233            executor, stale_output_key
234        );
235    }
236
237    fn reset_for_new_revision(&mut self) {
238        // Interned ingredients do not, normally, get deleted except when they are "reset" en masse.
239        // There ARE methods (e.g., `clear_deleted_entries` and `remove`) for deleting individual
240        // items, but those are only used for tracked struct ingredients.
241        panic!("unexpected call to `reset_for_new_revision`")
242    }
243
244    fn salsa_struct_deleted(&self, _db: &DB, _id: crate::Id) {
245        panic!("unexpected call: interned ingredients do not register for salsa struct deletion events");
246    }

248    fn fmt_index(&self, index: Option<crate::Id>, fmt: &mut fmt::Formatter<'_>) -> fmt::Result {
249        fmt_index(self.debug_name, index, fmt)
250    }
