25/// Created for each tracked struct.
26/// This ingredient only stores the "id" fields.
27/// It is a kind of "dressed up" interner;
28/// the active query + values of id fields are hashed to create the tracked struct id.
29/// The value fields are stored in [`crate::function::FunctionIngredient`] instances keyed by the tracked struct id.
30/// Unlike normal interners, tracked struct indices can be deleted and reused aggressively:
31/// when a tracked function re-executes,
32/// any tracked structs that it created before but did not create this time can be deleted.
33pub struct TrackedStructIngredient<Id, Data>
34where
35    Id: TrackedStructId,
36    Data: TrackedStructData,
37{
38    interned: InternedIngredient<Id, TrackedStructKey<Data>>,
39
40    /// A list of each tracked function `f` whose key is this
41    /// tracked struct.
42    ///
43    /// Whenever an instance `i` of this struct is deleted,
44    /// each of these functions will be notified
45    /// so they can remove any data tied to that instance.
46    dependent_fns: IngredientList,
47
48    debug_name: &'static str,
49}
50
51#[derive(Debug, PartialEq, Eq, Hash, PartialOrd, Ord, Copy, Clone)]
52struct TrackedStructKey<Data> {
53    query_key: Option<DatabaseKeyIndex>,
54    disambiguator: Disambiguator,
55    data: Data,
56}
57
58#[derive(Debug, PartialEq, Eq, Hash, PartialOrd, Ord, Copy, Clone)]
59pub struct Disambiguator(pub u32);
60
61impl<Id, Data> TrackedStructIngredient<Id, Data>
62where
63    Id: TrackedStructId,
64    Data: TrackedStructData,
65{
66    pub fn new(index: IngredientIndex, debug_name: &'static str) -> Self {
67        Self {
68            interned: InternedIngredient::new(index, debug_name),
69            dependent_fns: IngredientList::new(),
70            debug_name,
71        }
72    }
73
74    pub fn database_key_index(&self, id: Id) -> DatabaseKeyIndex {
75        DatabaseKeyIndex {
76            ingredient_index: self.interned.ingredient_index(),
77            key_index: id.as_id(),
78        }
79    }
80
81    pub fn new_struct(&self, runtime: &Runtime, data: Data) -> Id {
82        let data_hash = crate::hash::hash(&data);
83        let (query_key, disambiguator) = runtime.disambiguate_entity(
84            self.interned.ingredient_index(),
85            self.interned.reset_at(),
86            data_hash,
87        );
88        let entity_key = TrackedStructKey {
89            query_key: Some(query_key),
90            disambiguator,
91            data,
92        };
93        let result = self.interned.intern(runtime, entity_key);
94        runtime.add_output(self.database_key_index(result).into());
95        result
96    }
97
98    pub fn tracked_struct_data<'db>(&'db self, runtime: &'db Runtime, id: Id) -> &'db Data {
99        &self.interned.data(runtime, id).data
100    }
101
102    /// Deletes the given entities. This is used after a query `Q` executes and we can compare
103    /// the entities `E_now` that it produced in this revision vs the entities
104    /// `E_prev` it produced in the last revision. Any missing entities `E_prev - E_new` can be
105    /// deleted.
106    ///
107    /// # Warning
108    ///
109    /// Using this method on an entity id that MAY be used in the current revision will lead to
110    /// unspecified results (but not UB). See [`InternedIngredient::delete_index`] for more
111    /// discussion and important considerations.
112    pub(crate) fn delete_entity(&self, db: &dyn crate::Database, id: Id) {
113        db.salsa_event(Event {
114            runtime_id: db.runtime().id(),
115            kind: crate::EventKind::DidDiscard {
116                key: self.database_key_index(id),
117            },
118        });
119
120        self.interned.delete_index(id);
121        for dependent_fn in self.dependent_fns.iter() {
122            db.salsa_struct_deleted(dependent_fn, id.as_id());
123        }
124    }
125
126    /// Adds a dependent function (one keyed by this tracked struct) to our list.
127    /// When instances of this struct are deleted, these dependent functions
128    /// will be notified.
129    pub fn register_dependent_fn(&self, index: IngredientIndex) {
130        self.dependent_fns.push(index);
131    }
132}

134impl<DB: ?Sized, Id, Data> Ingredient<DB> for TrackedStructIngredient<Id, Data>
135where
136    Id: TrackedStructId,
137    Data: TrackedStructData,
138    DB: crate::Database,
139{
140    fn ingredient_index(&self) -> IngredientIndex {
141        self.interned.ingredient_index()
142    }
143
144    fn maybe_changed_after(&self, db: &DB, input: DependencyIndex, revision: Revision) -> bool {
145        self.interned.maybe_changed_after(db, input, revision)
146    }
147
148    fn cycle_recovery_strategy(&self) -> CycleRecoveryStrategy {
149        <_ as Ingredient<DB>>::cycle_recovery_strategy(&self.interned)
150    }
151
152    fn origin(&self, _key_index: crate::Id) -> Option<QueryOrigin> {
153        None
154    }
155
156    fn mark_validated_output(
157        &self,
158        _db: &DB,
159        _executor: DatabaseKeyIndex,
160        _output_key: Option<crate::Id>,
161    ) {
162        // FIXME
163    }
164
165    fn remove_stale_output(
166        &self,
167        db: &DB,
168        _executor: DatabaseKeyIndex,
169        stale_output_key: Option<crate::Id>,
170    ) {
171        // This method is called when, in prior revisions,
172        // `executor` creates a tracked struct `salsa_output_key`,
173        // but it did not in the current revision.
174        // In that case, we can delete `stale_output_key` and any data associated with it.
175        let stale_output_key: Id = Id::from_id(stale_output_key.unwrap());
176        self.delete_entity(db.as_salsa_database(), stale_output_key);
177    }
178
179    fn reset_for_new_revision(&mut self) {
180        self.interned.clear_deleted_indices();
181    }
182
183    fn salsa_struct_deleted(&self, _db: &DB, _id: crate::Id) {
184        panic!("unexpected call: interned ingredients do not register for salsa struct deletion events");
185    }
186
187    fn fmt_index(&self, index: Option<crate::Id>, fmt: &mut fmt::Formatter<'_>) -> fmt::Result {
188        fmt_index(self.debug_name, index, fmt)
189    }
190}
