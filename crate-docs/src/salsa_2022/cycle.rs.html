5/// Captures the participants of a cycle that occurred when executing a query.
6///
7/// This type is meant to be used to help give meaningful error messages to the
8/// user or to help salsa developers figure out why their program is resulting
9/// in a computation cycle.
10///
11/// It is used in a few ways:
12///
13/// * During [cycle recovery](https://https://salsa-rs.github.io/salsa/cycles/fallback.html),
14///   where it is given to the fallback function.
15/// * As the panic value when an unexpected cycle (i.e., a cycle where one or more participants
16///   lacks cycle recovery information) occurs.
17///
18/// You can read more about cycle handling in
19/// the [salsa book](https://https://salsa-rs.github.io/salsa/cycles.html).
20#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash)]
21pub struct Cycle {
22    participants: CycleParticipants,
23}
24
25pub(crate) type CycleParticipants = Arc<Vec<DatabaseKeyIndex>>;
26
27impl Cycle {
28    pub(crate) fn new(participants: CycleParticipants) -> Self {
29        Self { participants }
30    }
31
32    /// True if two `Cycle` values represent the same cycle.
33    pub(crate) fn is(&self, cycle: &Cycle) -> bool {
34        Arc::ptr_eq(&self.participants, &cycle.participants)
35    }
36
37    pub(crate) fn throw(self) -> ! {
38        log::debug!("throwing cycle {:?}", self);
39        std::panic::resume_unwind(Box::new(self))
40    }
41
42    pub(crate) fn catch<T>(execute: impl FnOnce() -> T) -> Result<T, Cycle> {
43        match std::panic::catch_unwind(AssertUnwindSafe(execute)) {
44            Ok(v) => Ok(v),
45            Err(err) => match err.downcast::<Cycle>() {
46                Ok(cycle) => Err(*cycle),
47                Err(other) => std::panic::resume_unwind(other),
48            },
49        }
50    }
51
52    /// Iterate over the [`DatabaseKeyIndex`] for each query participating
53    /// in the cycle. The start point of this iteration within the cycle
54    /// is arbitrary but deterministic, but the ordering is otherwise determined
55    /// by the execution.
56    pub fn participant_keys(&self) -> impl Iterator<Item = DatabaseKeyIndex> + '_ {
57        self.participants.iter().copied()
58    }
59
60    /// Returns a vector with the debug information for
61    /// all the participants in the cycle.
62    pub fn all_participants<DB: ?Sized + Database>(&self, db: &DB) -> Vec<String> {
63        self.participant_keys()
64            .map(|d| format!("{:?}", d.debug(db)))
65            .collect()
66    }
67
68    /// Returns a vector with the debug information for
69    /// those participants in the cycle that lacked recovery
70    /// information.
71    pub fn unexpected_participants<DB: ?Sized + Database>(&self, db: &DB) -> Vec<String> {
72        self.participant_keys()
73            .filter(|&d| {
74                db.cycle_recovery_strategy(d.ingredient_index) == CycleRecoveryStrategy::Panic
75            })
76            .map(|d| format!("{:?}", d.debug(db)))
77            .collect()
78    }
79
80    /// Returns a "debug" view onto this strict that can be used to print out information.
81    pub fn debug<'me, DB: ?Sized + Database>(&'me self, db: &'me DB) -> impl std::fmt::Debug + 'me {
82        struct UnexpectedCycleDebug<'me> {
83            c: &'me Cycle,
84            db: &'me dyn Database,
85        }
86
87        impl<'me> std::fmt::Debug for UnexpectedCycleDebug<'me> {
88            fn fmt(&self, fmt: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
89                fmt.debug_struct("UnexpectedCycle")
90                    .field("all_participants", &self.c.all_participants(self.db))
91                    .field(
92                        "unexpected_participants",
93                        &self.c.unexpected_participants(self.db),
94                    )
95                    .finish()
96            }
97        }
98
99        UnexpectedCycleDebug {
100            c: self,
101            db: db.as_salsa_database(),
102        }
103    }
104}
