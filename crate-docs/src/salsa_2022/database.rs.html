3pub trait Database: HasJarsDyn + AsSalsaDatabase {
4    /// This function is invoked at key points in the salsa
5    /// runtime. It permits the database to be customized and to
6    /// inject logging or other custom behavior.
7    ///
8    /// By default, the event is logged at level debug using
9    /// the standard `log` facade.
10    fn salsa_event(&self, event: Event) {
11        log::debug!("salsa_event: {:?}", event.debug(self));
12    }
13
14    /// A "synthetic write" causes the system to act *as though* some
15    /// input of durability `durability` has changed. This is mostly
16    /// useful for profiling scenarios.
17    ///
18    /// **WARNING:** Just like an ordinary write, this method triggers
19    /// cancellation. If you invoke it while a snapshot exists, it
20    /// will block until that snapshot is dropped -- if that snapshot
21    /// is owned by the current thread, this could trigger deadlock.
22    fn synthetic_write(&mut self, durability: Durability) {
23        self.runtime_mut().report_tracked_write(durability);
24    }
25
26    /// Reports that the query depends on some state unknown to salsa.
27    ///
28    /// Queries which report untracked reads will be re-executed in the next
29    /// revision.
30    fn report_untracked_read(&self) {
31        self.runtime().report_untracked_read();
32    }
33}

35/// Indicates a database that also supports parallel query
36/// evaluation. All of Salsa's base query support is capable of
37/// parallel execution, but for it to work, your query key/value types
38/// must also be `Send`, as must any additional data in your database.
39pub trait ParallelDatabase: Database + Send {
40    /// Creates a second handle to the database that holds the
41    /// database fixed at a particular revision. So long as this
42    /// "frozen" handle exists, any attempt to [`set`] an input will
43    /// block.
44    ///
45    /// [`set`]: struct.QueryTable.html#method.set
46    ///
47    /// This is the method you are meant to use most of the time in a
48    /// parallel setting where modifications may arise asynchronously
49    /// (e.g., a language server). In this context, it is common to
50    /// wish to "fork off" a snapshot of the database performing some
51    /// series of queries in parallel and arranging the results. Using
52    /// this method for that purpose ensures that those queries will
53    /// see a consistent view of the database (it is also advisable
54    /// for those queries to use the [`Runtime::unwind_if_cancelled`]
55    /// method to check for cancellation).
56    ///
57    /// # Panics
58    ///
59    /// It is not permitted to create a snapshot from inside of a
60    /// query. Attepting to do so will panic.
61    ///
62    /// # Deadlock warning
63    ///
64    /// The intended pattern for snapshots is that, once created, they
65    /// are sent to another thread and used from there. As such, the
66    /// `snapshot` acquires a "read lock" on the database --
67    /// therefore, so long as the `snapshot` is not dropped, any
68    /// attempt to `set` a value in the database will block. If the
69    /// `snapshot` is owned by the same thread that is attempting to
70    /// `set`, this will cause a problem.
71    ///
72    /// # How to implement this
73    ///
74    /// Typically, this method will create a second copy of your
75    /// database type (`MyDatabaseType`, in the example below),
76    /// cloning over each of the fields from `self` into this new
77    /// copy. For the field that stores the salsa runtime, you should
78    /// use [the `Runtime::snapshot` method][rfm] to create a snapshot of the
79    /// runtime. Finally, package up the result using `Snapshot::new`,
80    /// which is a simple wrapper type that only gives `&self` access
81    /// to the database within (thus preventing the use of methods
82    /// that may mutate the inputs):
83    ///
84    /// [rfm]: struct.Runtime.html#method.snapshot
85    ///
86    /// ```rust,ignore
87    /// impl ParallelDatabase for MyDatabaseType {
88    ///     fn snapshot(&self) -> Snapshot<Self> {
89    ///         Snapshot::new(
90    ///             MyDatabaseType {
91    ///                 runtime: self.storage.snapshot(),
92    ///                 other_field: self.other_field.clone(),
93    ///             }
94    ///         )
95    ///     }
96    /// }
97    /// ```
98    fn snapshot(&self) -> Snapshot<Self>;
99}
100pub trait AsSalsaDatabase {
101    fn as_salsa_database(&self) -> &dyn Database;
102}

104/// Simple wrapper struct that takes ownership of a database `DB` and
105/// only gives `&self` access to it. See [the `snapshot` method][fm]
106/// for more details.
107///
108/// [fm]: trait.ParallelDatabase.html#method.snapshot
109#[derive(Debug)]
110pub struct Snapshot<DB: ?Sized>
111where
112    DB: ParallelDatabase,
113{
114    db: DB,
115}

117impl<DB> Snapshot<DB>
118where

121    /// Creates a `Snapshot` that wraps the given database handle
122    /// `db`. From this point forward, only shared references to `db`

125        Snapshot { db }
126    }
