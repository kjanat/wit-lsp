16/// The "storage" struct stores all the data for the jars.
17/// It is shared between the main database and any active snapshots.
18pub struct Storage<DB: HasJars> {
19    /// Data shared across all databases. This contains the ingredients needed by each jar.
20    /// See the ["jars and ingredients" chapter](https://salsa-rs.github.io/salsa/plumbing/jars_and_ingredients.html)
21    /// for more detailed description.
22    shared: Shared<DB>,
23
24    /// The "ingredients" structure stores the information about how to find each ingredient in the database.
25    /// It allows us to take the [`IngredientIndex`] assigned to a particular ingredient
26    /// and get back a [`dyn Ingredient`][`Ingredient`] for the struct that stores its data.
27    ///
28    /// This is kept separate from `shared` so that we can clone it and retain `&`-access even when we have `&mut` access to `shared`.
29    routes: Arc<Routes<DB>>,
30
31    /// The runtime for this particular salsa database handle.
32    /// Each handle gets its own runtime, but the runtimes have shared state between them.
33    runtime: Runtime,
34}
35
36/// Data shared between all threads.
37/// This is where the actual data for tracked functions, structs, inputs, etc lives,
38/// along with some coordination variables between treads.
39struct Shared<DB: HasJars> {
40    /// Contains the data for each jar in the database.
41    /// Each jar stores its own structs in there that ultimately contain ingredients
42    /// (types that implement the [`Ingredient`] trait, like [`crate::function::FunctionIngredient`]).
43    ///
44    /// Even though these jars are stored in an `Arc`, we sometimes get mutable access to them
45    /// by using `Arc::get_mut`. This is only possible when all parallel snapshots have been dropped.
46    jars: Option<Arc<DB::Jars>>,
47
48    /// Conditional variable that is used to coordinate cancellation.
49    /// When the main thread writes to the database, it blocks until each of the snapshots can be cancelled.
50    cvar: Arc<Condvar>,
51}
52
53// ANCHOR: default
54impl<DB> Default for Storage<DB>
55where
56    DB: HasJars,
57{
58    fn default() -> Self {
59        let mut routes = Routes::new();
60        let jars = DB::create_jars(&mut routes);
61        Self {
62            shared: Shared {
63                jars: Some(Arc::from(jars)),
64                cvar: Arc::new(Default::default()),
65            },
66            routes: Arc::new(routes),
67            runtime: Runtime::default(),
68        }
69    }
70}

73impl<DB> Storage<DB>
74where
75    DB: HasJars,
76{
77    pub fn snapshot(&self) -> Storage<DB>
78    where
79        DB: ParallelDatabase,
80    {
81        Self {
82            shared: self.shared.clone(),
83            routes: self.routes.clone(),
84            runtime: self.runtime.snapshot(),
85        }
86    }
87
88    pub fn jars(&self) -> (&DB::Jars, &Runtime) {
89        (self.shared.jars.as_ref().unwrap(), &self.runtime)
90    }
91
92    pub fn runtime(&self) -> &Runtime {
93        &self.runtime
94    }
95
96    pub fn runtime_mut(&mut self) -> &mut Runtime {
97        self.jars_mut().1
98    }
99
100    // ANCHOR: jars_mut
101    /// Gets mutable access to the jars. This will trigger a new revision
102    /// and it will also cancel any ongoing work in the current revision.
103    /// Any actual writes that occur to data in a jar should use
104    /// [`Runtime::report_tracked_write`].
105    pub fn jars_mut(&mut self) -> (&mut DB::Jars, &mut Runtime) {
106        // Wait for all snapshots to be dropped.
107        self.cancel_other_workers();
108
109        // Increment revision counter.
110        self.runtime.new_revision();
111
112        // Acquire `&mut` access to `self.shared` -- this is only possible because
113        // the snapshots have all been dropped, so we hold the only handle to the `Arc`.
114        let jars = Arc::get_mut(self.shared.jars.as_mut().unwrap()).unwrap();
115
116        // Inform other ingredients that a new revision has begun.
117        // This gives them a chance to free resources that were being held until the next revision.
118        let routes = self.routes.clone();
119        for route in routes.reset_routes() {
120            route(jars).reset_for_new_revision();
121        }
122
123        // Return mut ref to jars + runtime.
124        (jars, &mut self.runtime)
125    }
126    // ANCHOR_END: jars_mut
127
128    // ANCHOR: cancel_other_workers
129    /// Sets cancellation flag and blocks until all other workers with access
130    /// to this storage have completed.
131    ///
132    /// This could deadlock if there is a single worker with two handles to the
133    /// same database!
134    fn cancel_other_workers(&mut self) {
135        loop {
136            self.runtime.set_cancellation_flag();
137
138            // If we have unique access to the jars, we are done.
139            if Arc::get_mut(self.shared.jars.as_mut().unwrap()).is_some() {
140                return;
141            }
142
143            // Otherwise, wait until some other storage entities have dropped.
144            // We create a mutex here because the cvar api requires it, but we
145            // don't really need one as the data being protected is actually
146            // the jars above.
147            //
148            // The cvar `self.shared.cvar` is notified by the `Drop` impl.
149            let mutex = parking_lot::Mutex::new(());
150            let mut guard = mutex.lock();
151            self.shared.cvar.wait(&mut guard);
152        }
153    }
154    // ANCHOR_END: cancel_other_workers
155
156    pub fn ingredient(&self, ingredient_index: IngredientIndex) -> &dyn Ingredient<DB> {
157        let route = self.routes.route(ingredient_index);
158        route(self.shared.jars.as_ref().unwrap())
159    }
160}

164    DB: HasJars,

171    }

174impl<DB> Drop for Storage<DB>
175where
176    DB: HasJars,
177{
178    fn drop(&mut self) {
179        // Drop the Arc reference before the cvar is notified,
180        // since other threads are sleeping, waiting for it to reach 1.
181        drop(self.shared.jars.take());
182        self.shared.cvar.notify_all();
183    }
184}
