45/// A unique identifier for a particular runtime. Each time you create

107    pub fn snapshot(&self) -> Self {
108        if self.local_state.query_in_progress() {
109            panic!("it is not legal to `snapshot` during a query (see salsa-rs/salsa#80)");
110        }
111
112        let id = RuntimeId {
113            counter: self.shared_state.next_id.fetch_add(1, Ordering::SeqCst),
114        };
115
116        Runtime {
117            id,
118            shared_state: self.shared_state.clone(),
119            local_state: Default::default(),
120        }
121    }

184    /// The revision in which values with durability `d` may have last
185    /// changed.  For D0, this is just the current revision. But for
186    /// higher levels of durability, this value may lag behind the
187    /// current revision. If we encounter a value of durability Di,
188    /// then, we can check this function to get a "bound" on when the
189    /// value may have changed, which allows us to skip walking its
190    /// dependencies.
191    #[inline]
192    pub(crate) fn last_changed_revision(&self, d: Durability) -> Revision {
193        self.shared_state.revisions[d.index()].load()
194    }

196    /// Starts unwinding the stack if the current revision is cancelled.
197    ///
198    /// This method can be called by query implementations that perform
199    /// potentially expensive computations, in order to speed up propagation of
200    /// cancellation.
201    ///
202    /// Cancellation will automatically be triggered by salsa on any query
203    /// invocation.
204    ///
205    /// This method should not be overridden by `Database` implementors. A
206    /// `salsa_event` is emitted when this method is called, so that should be
207    /// used instead.
208    pub(crate) fn unwind_if_revision_cancelled<DB: ?Sized + Database>(&self, db: &DB) {
209        db.salsa_event(Event {
210            runtime_id: self.id(),
211            kind: EventKind::WillCheckCancellation,
212        });
213        if self.shared_state.revision_canceled.load() {
214            db.salsa_event(Event {
215                runtime_id: self.id(),
216                kind: EventKind::WillCheckCancellation,
217            });
218            self.unwind_cancelled();
219        }
220    }

222    #[cold]
223    pub(crate) fn unwind_cancelled(&self) {
224        self.report_untracked_read();
225        Cancelled::PendingWrite.throw();

232    /// Increments the "current revision" counter and clears
233    /// the cancellation flag.
234    ///
235    /// This should only be done by the storage when the state is "quiescent".
236    pub(crate) fn new_revision(&mut self) -> Revision {
237        let r_old = self.current_revision();
238        let r_new = r_old.next();
239        self.shared_state.revisions[0].store(r_new);
240        self.shared_state.revision_canceled.store(false);
241        r_new
242    }

249    /// Block until `other_id` completes executing `database_key`;
250    /// panic or unwind in the case of a cycle.
251    ///
252    /// `query_mutex_guard` is the guard for the current query's state;
253    /// it will be dropped after we have successfully registered the
254    /// dependency.
255    ///
256    /// # Propagating panics
257    ///
258    /// If the thread `other_id` panics, then our thread is considered
259    /// cancelled, so this function will panic with a `Cancelled` value.
260    ///
261    /// # Cycle handling
262    ///
263    /// If the thread `other_id` already depends on the current thread,
264    /// and hence there is a cycle in the query graph, then this function
265    /// will unwind instead of returning normally. The method of unwinding
266    /// depends on the [`Self::mutual_cycle_recovery_strategy`]
267    /// of the cycle participants:
268    ///
269    /// * [`CycleRecoveryStrategy::Panic`]: panic with the [`Cycle`] as the value.
270    /// * [`CycleRecoveryStrategy::Fallback`]: initiate unwinding with [`CycleParticipant::unwind`].
271    pub(crate) fn block_on_or_unwind<QueryMutexGuard>(
272        &self,
273        db: &dyn Database,
274        database_key: DatabaseKeyIndex,
275        other_id: RuntimeId,
276        query_mutex_guard: QueryMutexGuard,
277    ) {
278        let mut dg = self.shared_state.dependency_graph.lock();
279
280        if dg.depends_on(other_id, self.id()) {
281            self.unblock_cycle_and_maybe_throw(db, &mut dg, database_key, other_id);
282
283            // If the above fn returns, then (via cycle recovery) it has unblocked the
284            // cycle, so we can continue.
285            assert!(!dg.depends_on(other_id, self.id()));
286        }
287
288        db.salsa_event(Event {
289            runtime_id: self.id(),
290            kind: EventKind::WillBlockOn {
291                other_runtime_id: other_id,
292                database_key,
293            },
294        });
295
296        let stack = self.local_state.take_query_stack();
297
298        let (stack, result) = DependencyGraph::block_on(
299            dg,
300            self.id(),
301            database_key,
302            other_id,
303            stack,
304            query_mutex_guard,
305        );
306
307        self.local_state.restore_query_stack(stack);
308
309        match result {
310            WaitResult::Completed => (),
311
312            // If the other thread panicked, then we consider this thread
313            // cancelled. The assumption is that the panic will be detected
314            // by the other thread and responded to appropriately.
315            WaitResult::Panicked => Cancelled::PropagatedPanic.throw(),
316
317            WaitResult::Cycle(c) => c.throw(),
318        }
319    }

321    /// Handles a cycle in the dependency graph that was detected when the
322    /// current thread tried to block on `database_key_index` which is being
323    /// executed by `to_id`. If this function returns, then `to_id` no longer
324    /// depends on the current thread, and so we should continue executing
325    /// as normal. Otherwise, the function will throw a `Cycle` which is expected
326    /// to be caught by some frame on our stack. This occurs either if there is
327    /// a frame on our stack with cycle recovery (possibly the top one!) or if there
328    /// is no cycle recovery at all.
329    fn unblock_cycle_and_maybe_throw(
330        &self,
331        db: &dyn Database,
332        dg: &mut DependencyGraph,
333        database_key_index: DatabaseKeyIndex,
334        to_id: RuntimeId,
335    ) {
336        log::debug!(
337            "unblock_cycle_and_maybe_throw(database_key={:?})",
338            database_key_index
339        );
340
341        let mut from_stack = self.local_state.take_query_stack();
342        let from_id = self.id();
343
344        // Make a "dummy stack frame". As we iterate through the cycle, we will collect the
345        // inputs from each participant. Then, if we are participating in cycle recovery, we
346        // will propagate those results to all participants.
347        let mut cycle_query = ActiveQuery::new(database_key_index);
348
349        // Identify the cycle participants:
350        let cycle = {
351            let mut v = vec![];
352            dg.for_each_cycle_participant(
353                from_id,
354                &mut from_stack,
355                database_key_index,
356                to_id,
357                |aqs| {
358                    aqs.iter_mut().for_each(|aq| {
359                        cycle_query.add_from(aq);
360                        v.push(aq.database_key_index);
361                    });
362                },
363            );
364
365            // We want to give the participants in a deterministic order
366            // (at least for this execution, not necessarily across executions),
367            // no matter where it started on the stack. Find the minimum
368            // key and rotate it to the front.
369            let min = v.iter().min().unwrap();
370            let index = v.iter().position(|p| p == min).unwrap();
371            v.rotate_left(index);
372
373            // No need to store extra memory.
374            v.shrink_to_fit();
375
376            Cycle::new(Arc::new(v))
377        };
378        log::debug!(
379            "cycle {:?}, cycle_query {:#?}",
380            cycle.debug(db),
381            cycle_query,
382        );
383
384        // We can remove the cycle participants from the list of dependencies;
385        // they are a strongly connected component (SCC) and we only care about
386        // dependencies to things outside the SCC that control whether it will
387        // form again.
388        cycle_query.remove_cycle_participants(&cycle);
389
390        // Mark each cycle participant that has recovery set, along with
391        // any frames that come after them on the same thread. Those frames
392        // are going to be unwound so that fallback can occur.
393        dg.for_each_cycle_participant(from_id, &mut from_stack, database_key_index, to_id, |aqs| {
394            aqs.iter_mut()
395                .skip_while(|aq| {
396                    match db.cycle_recovery_strategy(aq.database_key_index.ingredient_index) {
397                        CycleRecoveryStrategy::Panic => true,
398                        CycleRecoveryStrategy::Fallback => false,
399                    }
400                })
401                .for_each(|aq| {
402                    log::debug!("marking {:?} for fallback", aq.database_key_index.debug(db));
403                    aq.take_inputs_from(&cycle_query);
404                    assert!(aq.cycle.is_none());
405                    aq.cycle = Some(cycle.clone());
406                });
407        });
408
409        // Unblock every thread that has cycle recovery with a `WaitResult::Cycle`.
410        // They will throw the cycle, which will be caught by the frame that has
411        // cycle recovery so that it can execute that recovery.
412        let (me_recovered, others_recovered) =
413            dg.maybe_unblock_runtimes_in_cycle(from_id, &from_stack, database_key_index, to_id);
414
415        self.local_state.restore_query_stack(from_stack);
416
417        if me_recovered {
418            // If the current thread has recovery, we want to throw
419            // so that it can begin.
420            cycle.throw()
421        } else if others_recovered {
422            // If other threads have recovery but we didn't: return and we will block on them.
423        } else {
424            // if nobody has recover, then we panic
425            panic_any(cycle);
426        }
427    }

429    /// Invoked when this runtime completed computing `database_key` with
430    /// the given result `wait_result` (`wait_result` should be `None` if
431    /// computing `database_key` panicked and could not complete).
432    /// This function unblocks any dependent queries and allows them
433    /// to continue executing.
434    pub(crate) fn unblock_queries_blocked_on(
435        &self,
436        database_key: DatabaseKeyIndex,
437        wait_result: WaitResult,
438    ) {
439        self.shared_state
440            .dependency_graph
441            .lock()
442            .unblock_runtimes_blocked_on(database_key, wait_result);
443    }
