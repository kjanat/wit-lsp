6/// The `Event` struct identifies various notable things that can
7/// occur during salsa execution. Instances of this struct are given
8/// to `salsa_event`.
9pub struct Event {
10    /// The id of the snapshot that triggered the event.  Usually
11    /// 1-to-1 with a thread, as well.
12    pub runtime_id: RuntimeId,
13
14    /// What sort of event was it.
15    pub kind: EventKind,
16}
17
18impl fmt::Debug for Event {
19    fn fmt(&self, fmt: &mut fmt::Formatter<'_>) -> fmt::Result {
20        fmt.debug_struct("Event")
21            .field("runtime_id", &self.runtime_id)
22            .field("kind", &self.kind)
23            .finish()
24    }
25}

44/// An enum identifying the various kinds of events that can occur.
45pub enum EventKind {
46    /// Occurs when we found that all inputs to a memoized value are
47    /// up-to-date and hence the value can be re-used without
48    /// executing the closure.
49    ///
50    /// Executes before the "re-used" value is returned.
51    DidValidateMemoizedValue {
52        /// The database-key for the affected value. Implements `Debug`.
53        database_key: DatabaseKeyIndex,
54    },
55
56    /// Indicates that another thread (with id `other_runtime_id`) is processing the
57    /// given query (`database_key`), so we will block until they
58    /// finish.
59    ///
60    /// Executes after we have registered with the other thread but
61    /// before they have answered us.
62    ///
63    /// (NB: you can find the `id` of the current thread via the
64    /// `runtime`)
65    WillBlockOn {
66        /// The id of the runtime we will block on.
67        other_runtime_id: RuntimeId,
68
69        /// The database-key for the affected value. Implements `Debug`.
70        database_key: DatabaseKeyIndex,
71    },
72
73    /// Indicates that the function for this query will be executed.
74    /// This is either because it has never executed before or because
75    /// its inputs may be out of date.
76    WillExecute {
77        /// The database-key for the affected value. Implements `Debug`.
78        database_key: DatabaseKeyIndex,
79    },
80
81    /// Indicates that `unwind_if_cancelled` was called and salsa will check if
82    /// the current revision has been cancelled.
83    WillCheckCancellation,
84
85    /// Discovered that a query used to output a given output but no longer does.
86    WillDiscardStaleOutput {
87        /// Key for the query that is executing and which no longer outputs the given value.
88        execute_key: DatabaseKeyIndex,
89
90        /// Key for the query that is no longer output
91        output_key: DependencyIndex,
92    },
93
94    /// Tracked structs or memoized data were discarded (freed).
95    DidDiscard {
96        /// Value being discarded.
97        key: DatabaseKeyIndex,
98    },
99
100    /// Discarded accumulated data from a given fn
101    DidDiscardAccumulated {
102        /// The key of the fn that accumulated results
103        executor_key: DatabaseKeyIndex,
104
105        /// Accumulator that was accumulated into
106        accumulator: DependencyIndex,
107    },
108}
109
110impl fmt::Debug for EventKind {
111    fn fmt(&self, fmt: &mut fmt::Formatter<'_>) -> fmt::Result {
112        match self {
113            EventKind::DidValidateMemoizedValue { database_key } => fmt
114                .debug_struct("DidValidateMemoizedValue")
115                .field("database_key", database_key)
116                .finish(),
117            EventKind::WillBlockOn {
118                other_runtime_id,
119                database_key,
120            } => fmt
121                .debug_struct("WillBlockOn")
122                .field("other_runtime_id", other_runtime_id)
123                .field("database_key", database_key)
124                .finish(),
125            EventKind::WillExecute { database_key } => fmt
126                .debug_struct("WillExecute")
127                .field("database_key", database_key)
128                .finish(),
129            EventKind::WillCheckCancellation => fmt.debug_struct("WillCheckCancellation").finish(),
130            EventKind::WillDiscardStaleOutput {
131                execute_key,
132                output_key,
133            } => fmt
134                .debug_struct("WillDiscardStaleOutput")
135                .field("execute_key", &execute_key)
136                .field("output_key", &output_key)
137                .finish(),
138            EventKind::DidDiscard { key } => {
139                fmt.debug_struct("DidDiscard").field("key", &key).finish()
140            }
141            EventKind::DidDiscardAccumulated {
142                executor_key,
143                accumulator,
144            } => fmt
145                .debug_struct("DidDiscardAccumulated")
146                .field("executor_key", executor_key)
147                .field("accumulator", accumulator)
148                .finish(),
149        }
150    }
151}
