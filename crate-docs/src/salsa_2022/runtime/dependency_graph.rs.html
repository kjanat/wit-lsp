3use crate::key::DatabaseKeyIndex;
4use parking_lot::{Condvar, MutexGuard};
5use rustc_hash::FxHashMap;
6use smallvec::SmallVec;

8use super::{active_query::ActiveQuery, RuntimeId, WaitResult};

10type QueryStack = Vec<ActiveQuery>;
11
12#[derive(Debug, Default)]
13pub(super) struct DependencyGraph {
14    /// A `(K -> V)` pair in this map indicates that the the runtime
15    /// `K` is blocked on some query executing in the runtime `V`.
16    /// This encodes a graph that must be acyclic (or else deadlock
17    /// will result).
18    edges: FxHashMap<RuntimeId, Edge>,
19
20    /// Encodes the `RuntimeId` that are blocked waiting for the result
21    /// of a given query.
22    query_dependents: FxHashMap<DatabaseKeyIndex, SmallVec<[RuntimeId; 4]>>,
23
24    /// When a key K completes which had dependent queries Qs blocked on it,
25    /// it stores its `WaitResult` here. As they wake up, each query Q in Qs will
26    /// come here to fetch their results.
27    wait_results: FxHashMap<RuntimeId, (QueryStack, WaitResult)>,
28}
29
30#[derive(Debug)]
31struct Edge {
32    blocked_on_id: RuntimeId,
33    blocked_on_key: DatabaseKeyIndex,
34    stack: QueryStack,
35
36    /// Signalled whenever a query with dependents completes.
37    /// Allows those dependents to check if they are ready to unblock.
38    condvar: Arc<parking_lot::Condvar>,
39}
40
41impl DependencyGraph {
42    /// True if `from_id` depends on `to_id`.
43    ///
44    /// (i.e., there is a path from `from_id` to `to_id` in the graph.)
45    pub(super) fn depends_on(&mut self, from_id: RuntimeId, to_id: RuntimeId) -> bool {
46        let mut p = from_id;
47        while let Some(q) = self.edges.get(&p).map(|edge| edge.blocked_on_id) {
48            if q == to_id {
49                return true;
50            }
51
52            p = q;
53        }
54        p == to_id
55    }
56
57    /// Invokes `closure` with a `&mut ActiveQuery` for each query that participates in the cycle.
58    /// The cycle runs as follows:
59    ///
60    /// 1. The runtime `from_id`, which has the stack `from_stack`, would like to invoke `database_key`...
61    /// 2. ...but `database_key` is already being executed by `to_id`...
62    /// 3. ...and `to_id` is transitively dependent on something which is present on `from_stack`.
63    pub(super) fn for_each_cycle_participant(
64        &mut self,
65        from_id: RuntimeId,
66        from_stack: &mut QueryStack,
67        database_key: DatabaseKeyIndex,
68        to_id: RuntimeId,
69        mut closure: impl FnMut(&mut [ActiveQuery]),
70    ) {
71        debug_assert!(self.depends_on(to_id, from_id));
72
73        // To understand this algorithm, consider this [drawing](https://is.gd/TGLI9v):
74        //
75        //    database_key = QB2
76        //    from_id = A
77        //    to_id = B
78        //    from_stack = [QA1, QA2, QA3]
79        //
80        //    self.edges[B] = { C, QC2, [QB1..QB3] }
81        //    self.edges[C] = { A, QA2, [QC1..QC3] }
82        //
83        //         The cyclic
84        //         edge we have
85        //         failed to add.
86        //           :
87        //    A      :    B         C
88        //           :
89        //    QA1    v    QB1       QC1
90        // ┌► QA2    ┌──► QB2   ┌─► QC2
91        // │  QA3 ───┘    QB3 ──┘   QC3 ───┐
92        // │                               │
93        // └───────────────────────────────┘
94        //
95        // Final output: [QB2, QB3, QC2, QC3, QA2, QA3]
96
97        let mut id = to_id;
98        let mut key = database_key;
99        while id != from_id {
100            // Looking at the diagram above, the idea is to
101            // take the edge from `to_id` starting at `key`
102            // (inclusive) and down to the end. We can then
103            // load up the next thread (i.e., we start at B/QB2,
104            // and then load up the dependency on C/QC2).
105            let edge = self.edges.get_mut(&id).unwrap();
106            let prefix = edge
107                .stack
108                .iter_mut()
109                .take_while(|p| p.database_key_index != key)
110                .count();
111            closure(&mut edge.stack[prefix..]);
112            id = edge.blocked_on_id;
113            key = edge.blocked_on_key;
114        }
115
116        // Finally, we copy in the results from `from_stack`.
117        let prefix = from_stack
118            .iter_mut()
119            .take_while(|p| p.database_key_index != key)
120            .count();
121        closure(&mut from_stack[prefix..]);
122    }
123
124    /// Unblock each blocked runtime (excluding the current one) if some
125    /// query executing in that runtime is participating in cycle fallback.
126    ///
127    /// Returns a boolean (Current, Others) where:
128    /// * Current is true if the current runtime has cycle participants
129    ///   with fallback;
130    /// * Others is true if other runtimes were unblocked.
131    pub(super) fn maybe_unblock_runtimes_in_cycle(
132        &mut self,
133        from_id: RuntimeId,
134        from_stack: &QueryStack,
135        database_key: DatabaseKeyIndex,
136        to_id: RuntimeId,
137    ) -> (bool, bool) {
138        // See diagram in `for_each_cycle_participant`.
139        let mut id = to_id;
140        let mut key = database_key;
141        let mut others_unblocked = false;
142        while id != from_id {
143            let edge = self.edges.get(&id).unwrap();
144            let prefix = edge
145                .stack
146                .iter()
147                .take_while(|p| p.database_key_index != key)
148                .count();
149            let next_id = edge.blocked_on_id;
150            let next_key = edge.blocked_on_key;
151
152            if let Some(cycle) = edge.stack[prefix..]
153                .iter()
154                .rev()
155                .find_map(|aq| aq.cycle.clone())
156            {
157                // Remove `id` from the list of runtimes blocked on `next_key`:
158                self.query_dependents
159                    .get_mut(&next_key)
160                    .unwrap()
161                    .retain(|r| *r != id);
162
163                // Unblock runtime so that it can resume execution once lock is released:
164                self.unblock_runtime(id, WaitResult::Cycle(cycle));
165
166                others_unblocked = true;
167            }
168
169            id = next_id;
170            key = next_key;
171        }
172
173        let prefix = from_stack
174            .iter()
175            .take_while(|p| p.database_key_index != key)
176            .count();
177        let this_unblocked = from_stack[prefix..].iter().any(|aq| aq.cycle.is_some());
178
179        (this_unblocked, others_unblocked)
180    }
181
182    /// Modifies the graph so that `from_id` is blocked
183    /// on `database_key`, which is being computed by
184    /// `to_id`.
185    ///
186    /// For this to be reasonable, the lock on the
187    /// results table for `database_key` must be held.
188    /// This ensures that computing `database_key` doesn't
189    /// complete before `block_on` executes.
190    ///
191    /// Preconditions:
192    /// * No path from `to_id` to `from_id`
193    ///   (i.e., `me.depends_on(to_id, from_id)` is false)
194    /// * `held_mutex` is a read lock (or stronger) on `database_key`
195    pub(super) fn block_on<QueryMutexGuard>(
196        mut me: MutexGuard<'_, Self>,
197        from_id: RuntimeId,
198        database_key: DatabaseKeyIndex,
199        to_id: RuntimeId,
200        from_stack: QueryStack,
201        query_mutex_guard: QueryMutexGuard,
202    ) -> (QueryStack, WaitResult) {
203        let condvar = me.add_edge(from_id, database_key, to_id, from_stack);
204
205        // Release the mutex that prevents `database_key`
206        // from completing, now that the edge has been added.
207        drop(query_mutex_guard);
208
209        loop {
210            if let Some(stack_and_result) = me.wait_results.remove(&from_id) {
211                debug_assert!(!me.edges.contains_key(&from_id));
212                return stack_and_result;
213            }
214            condvar.wait(&mut me);
215        }
216    }
217
218    /// Helper for `block_on`: performs actual graph modification
219    /// to add a dependency edge from `from_id` to `to_id`, which is
220    /// computing `database_key`.
221    fn add_edge(
222        &mut self,
223        from_id: RuntimeId,
224        database_key: DatabaseKeyIndex,
225        to_id: RuntimeId,
226        from_stack: QueryStack,
227    ) -> Arc<parking_lot::Condvar> {
228        assert_ne!(from_id, to_id);
229        debug_assert!(!self.edges.contains_key(&from_id));
230        debug_assert!(!self.depends_on(to_id, from_id));
231
232        let condvar = Arc::new(Condvar::new());
233        self.edges.insert(
234            from_id,
235            Edge {
236                blocked_on_id: to_id,
237                blocked_on_key: database_key,
238                stack: from_stack,
239                condvar: condvar.clone(),
240            },
241        );
242        self.query_dependents
243            .entry(database_key)
244            .or_default()
245            .push(from_id);
246        condvar
247    }
248
249    /// Invoked when runtime `to_id` completes executing
250    /// `database_key`.
251    pub(super) fn unblock_runtimes_blocked_on(
252        &mut self,
253        database_key: DatabaseKeyIndex,
254        wait_result: WaitResult,
255    ) {
256        let dependents = self
257            .query_dependents
258            .remove(&database_key)
259            .unwrap_or_default();
260
261        for from_id in dependents {
262            self.unblock_runtime(from_id, wait_result.clone());
263        }
264    }
265
266    /// Unblock the runtime with the given id with the given wait-result.
267    /// This will cause it resume execution (though it will have to grab
268    /// the lock on this data structure first, to recover the wait result).
269    fn unblock_runtime(&mut self, id: RuntimeId, wait_result: WaitResult) {
270        let edge = self.edges.remove(&id).expect("not blocked");
271        self.wait_results.insert(id, (edge.stack, wait_result));
272
273        // Now that we have inserted the `wait_results`,
274        // notify the thread.
275        edge.condvar.notify_one();
276    }
277}
