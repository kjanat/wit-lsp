21    pub(super) fn maybe_changed_after(
22        &self,
23        db: &DynDb<C>,
24        key: C::Key,
25        revision: Revision,
26    ) -> bool {
27        let runtime = db.runtime();
28        runtime.unwind_if_revision_cancelled(db);
29
30        loop {
31            let database_key_index = self.database_key_index(key);
32
33            log::debug!(
34                "{:?}: maybe_changed_after(revision = {:?})",
35                database_key_index.debug(db),
36                revision,
37            );
38
39            // Check if we have a verified version: this is the hot path.
40            let memo_guard = self.memo_map.get(key);
41            if let Some(memo) = &memo_guard {
42                if self.shallow_verify_memo(db, runtime, database_key_index, memo) {
43                    return memo.revisions.changed_at > revision;
44                }
45                drop(memo_guard); // release the arc-swap guard before cold path
46                if let Some(mcs) = self.maybe_changed_after_cold(db, key, revision) {
47                    return mcs;
48                } else {
49                    // We failed to claim, have to retry.
50                }
51            } else {
52                // No memo? Assume has changed.
53                return true;
54            }
55        }
56    }

58    fn maybe_changed_after_cold(
59        &self,
60        db: &DynDb<C>,
61        key_index: C::Key,
62        revision: Revision,
63    ) -> Option<bool> {
64        let runtime = db.runtime();
65        let database_key_index = self.database_key_index(key_index);
66
67        let _claim_guard = self
68            .sync_map
69            .claim(db.as_salsa_database(), database_key_index)?;
70        let active_query = runtime.push_query(database_key_index);
71
72        // Load the current memo, if any. Use a real arc, not an arc-swap guard,
73        // since we may recurse.
74        let old_memo = match self.memo_map.get(key_index) {
75            Some(m) => Guard::into_inner(m),
76            None => return Some(true),
77        };
78
79        log::debug!(
80            "{:?}: maybe_changed_after_cold, successful claim, revision = {:?}, old_memo = {:#?}",
81            database_key_index.debug(db),
82            revision,
83            old_memo
84        );
85
86        // Check if the inputs are still valid and we can just compare `changed_at`.
87        if self.deep_verify_memo(db, &old_memo, &active_query) {
88            return Some(old_memo.revisions.changed_at > revision);
89        }
90
91        // If inputs have changed, but we have an old value, we can re-execute.
92        // It is possible the result will be equal to the old value and hence
93        // backdated. In that case, although we will have computed a new memo,
94        // the value has not logically changed.
95        if old_memo.value.is_some() {
96            let StampedValue { changed_at, .. } = self.execute(db, active_query, Some(old_memo));
97            return Some(changed_at > revision);
98        }
99
100        // Otherwise, nothing for it: have to consider the value to have changed.
101        Some(true)
102    }

104    /// True if the memo's value and `changed_at` time is still valid in this revision.
105    /// Does only a shallow O(1) check, doesn't walk the dependencies.
106    #[inline]
107    pub(super) fn shallow_verify_memo(
108        &self,
109        db: &DynDb<C>,
110        runtime: &Runtime,
111        database_key_index: DatabaseKeyIndex,
112        memo: &Memo<C::Value>,
113    ) -> bool {
114        let verified_at = memo.verified_at.load();
115        let revision_now = runtime.current_revision();
116
117        log::debug!(
118            "{:?}: shallow_verify_memo(memo = {:#?})",
119            database_key_index.debug(db),
120            memo,
121        );
122
123        if verified_at == revision_now {
124            // Already verified.
125            return true;
126        }
127
128        if memo.check_durability(runtime) {
129            // No input of the suitable durability has changed since last verified.
130            memo.mark_as_verified(db.as_salsa_database(), runtime, database_key_index);
131            return true;
132        }
133
134        false
135    }

137    /// True if the memo's value and `changed_at` time is up to date in the current
138    /// revision. When this returns true, it also updates the memo's `verified_at`
139    /// field if needed to make future calls cheaper.
140    ///
141    /// Takes an [`ActiveQueryGuard`] argument because this function recursively
142    /// walks dependencies of `old_memo` and may even execute them to see if their
143    /// outputs have changed. As that could lead to cycles, it is important that the
144    /// query is on the stack.
145    pub(super) fn deep_verify_memo(
146        &self,
147        db: &DynDb<C>,
148        old_memo: &Memo<C::Value>,
149        active_query: &ActiveQueryGuard<'_>,
150    ) -> bool {
151        let runtime = db.runtime();
152        let database_key_index = active_query.database_key_index;
153
154        log::debug!(
155            "{:?}: deep_verify_memo(old_memo = {:#?})",
156            database_key_index.debug(db),
157            old_memo
158        );
159
160        if self.shallow_verify_memo(db, runtime, database_key_index, old_memo) {
161            return true;
162        }
163
164        match &old_memo.revisions.origin {
165            QueryOrigin::Assigned(_) => {
166                // If the value was assigneed by another query,
167                // and that query were up-to-date,
168                // then we would have updated the `verified_at` field already.
169                // So the fact that we are here means that it was not specified
170                // during this revision or is otherwise stale.
171                return false;
172            }
173            QueryOrigin::BaseInput => {
174                // This value was `set` by the mutator thread -- ie, it's a base input and it cannot be out of date.
175                return true;
176            }
177            QueryOrigin::DerivedUntracked(_) => {
178                // Untracked inputs? Have to assume that it changed.
179                return false;
180            }
181            QueryOrigin::Derived(edges) => {
182                // Fully tracked inputs? Iterate over the inputs and check them, one by one.
183                //
184                // NB: It's important here that we are iterating the inputs in the order that
185                // they executed. It's possible that if the value of some input I0 is no longer
186                // valid, then some later input I1 might never have executed at all, so verifying
187                // it is still up to date is meaningless.
188                let last_verified_at = old_memo.verified_at.load();
189                for &(edge_kind, dependency_index) in edges.input_outputs.iter() {
190                    match edge_kind {
191                        EdgeKind::Input => {
192                            if db.maybe_changed_after(dependency_index, last_verified_at) {
193                                return false;
194                            }
195                        }
196                        EdgeKind::Output => {
197                            // Subtle: Mark outputs as validated now, even though we may
198                            // later find an input that requires us to re-execute the function.
199                            // Even if it re-execute, the function will wind up writing the same value,
200                            // since all prior inputs were green. It's important to do this during
201                            // this loop, because it's possible that one of our input queries will
202                            // re-execute and may read one of our earlier outputs
203                            // (e.g., in a scenario where we do something like
204                            // `e = Entity::new(..); query(e);` and `query` reads a field of `e`).
205                            //
206                            // NB. Accumulators are also outputs, but the above logic doesn't
207                            // quite apply to them. Since multiple values are pushed, the first value
208                            // may be unchanged, but later values could be different.
209                            // In that case, however, the data accumulated
210                            // by this function cannot be read until this function is marked green,
211                            // so even if we mark them as valid here, the function will re-execute
212                            // and overwrite the contents.
213                            db.mark_validated_output(database_key_index, dependency_index);
214                        }
215                    }
216                }
217            }
218        }
219
220        old_memo.mark_as_verified(db.as_salsa_database(), runtime, database_key_index);
221        true
222    }
