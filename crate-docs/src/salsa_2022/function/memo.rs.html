11/// The memo map maps from a key of type `K` to the memoized value for that `K`.
12/// The memoized value is a `Memo<V>` which contains, in addition to the value `V`,
13/// dependency information.
14pub(super) struct MemoMap<K: AsId, V> {
15    map: FxDashMap<K, ArcSwap<Memo<V>>>,
16}
17
18impl<K: AsId, V> Default for MemoMap<K, V> {
19    fn default() -> Self {
20        Self {
21            map: Default::default(),
22        }
23    }
24}
25
26impl<K: AsId, V> MemoMap<K, V> {
27    /// Inserts the memo for the given key; (atomically) overwrites any previously existing memo.-
28    #[must_use]
29    pub(super) fn insert(&self, key: K, memo: Arc<Memo<V>>) -> Option<ArcSwap<Memo<V>>> {
30        self.map.insert(key, ArcSwap::from(memo))
31    }
32
33    /// Removes any existing memo for the given key.
34    #[must_use]
35    pub(super) fn remove(&self, key: K) -> Option<ArcSwap<Memo<V>>> {
36        self.map.remove(&key).map(|o| o.1)
37    }
38
39    /// Loads the current memo for `key_index`. This does not hold any sort of
40    /// lock on the `memo_map` once it returns, so this memo could immediately
41    /// become outdated if other threads store into the `memo_map`.
42    pub(super) fn get(&self, key: K) -> Option<Guard<Arc<Memo<V>>>> {
43        self.map.get(&key).map(|v| v.load())
44    }
45
46    /// Evicts the existing memo for the given key, replacing it
47    /// with an equivalent memo that has no value. If the memo is untracked, BaseInput,
48    /// or has values assigned as output of another query, this has no effect.
49    pub(super) fn evict(&self, key: K) {
50        use crate::runtime::local_state::QueryOrigin;
51        use dashmap::mapref::entry::Entry::*;
52
53        if let Occupied(entry) = self.map.entry(key) {
54            let memo = entry.get().load();
55            match memo.revisions.origin {
56                QueryOrigin::Assigned(_)
57                | QueryOrigin::DerivedUntracked(_)
58                | QueryOrigin::BaseInput => {
59                    // Careful: Cannot evict memos whose values were
60                    // assigned as output of another query
61                    // or those with untracked inputs
62                    // as their values cannot be reconstructed.
63                }
64
65                QueryOrigin::Derived(_) => {
66                    let memo_evicted = Arc::new(Memo::new(
67                        None::<V>,
68                        memo.verified_at.load(),
69                        memo.revisions.clone(),
70                    ));
71
72                    entry.get().store(memo_evicted);
73                }
74            }
75        }
76    }
77}

79#[derive(Debug)]
80pub(super) struct Memo<V> {
81    /// The result of the query, if we decide to memoize it.
82    pub(super) value: Option<V>,
83
84    /// Last revision when this memo was verified; this begins
85    /// as the current revision.
86    pub(super) verified_at: AtomicCell<Revision>,
87
88    /// Revision information
89    pub(super) revisions: QueryRevisions,
90}
91
92impl<V> Memo<V> {
93    pub(super) fn new(value: Option<V>, revision_now: Revision, revisions: QueryRevisions) -> Self {
94        Memo {
95            value,
96            verified_at: AtomicCell::new(revision_now),
97            revisions,
98        }
99    }
100    /// True if this memo is known not to have changed based on its durability.
101    pub(super) fn check_durability(&self, runtime: &Runtime) -> bool {
102        let last_changed = runtime.last_changed_revision(self.revisions.durability);
103        let verified_at = self.verified_at.load();
104        log::debug!(
105            "check_durability(last_changed={:?} <= verified_at={:?}) = {:?}",
106            last_changed,
107            self.verified_at,
108            last_changed <= verified_at,
109        );
110        last_changed <= verified_at
111    }
112
113    /// Mark memo as having been verified in the `revision_now`, which should
114    /// be the current revision.
115    pub(super) fn mark_as_verified(
116        &self,
117        db: &dyn crate::Database,
118        runtime: &crate::Runtime,
119        database_key_index: DatabaseKeyIndex,
120    ) {
121        db.salsa_event(Event {
122            runtime_id: runtime.id(),
123            kind: EventKind::DidValidateMemoizedValue {
124                database_key: database_key_index,
125            },
126        });
127
128        self.verified_at.store(runtime.current_revision());
129
130        // Also mark the outputs as verified
131        for output in self.revisions.origin.outputs() {
132            db.mark_validated_output(database_key_index, output);
133        }
134    }
135}
