11
12impl<C> FunctionIngredient<C>
13where
14    C: Configuration,
15{
16    /// Executes the query function for the given `active_query`. Creates and stores
17    /// a new memo with the result, backdated if possible. Once this completes,
18    /// the query will have been popped off the active query stack.
19    ///
20    /// # Parameters
21    ///
22    /// * `db`, the database.
23    /// * `active_query`, the active stack frame for the query to execute.
24    /// * `opt_old_memo`, the older memo, if any existed. Used for backdated.
25    pub(super) fn execute(
26        &self,
27        db: &DynDb<C>,
28        active_query: ActiveQueryGuard<'_>,
29        opt_old_memo: Option<Arc<Memo<C::Value>>>,
30    ) -> StampedValue<&C::Value> {
31        let runtime = db.runtime();
32        let revision_now = runtime.current_revision();
33        let database_key_index = active_query.database_key_index;
34
35        log::info!("{:?}: executing query", database_key_index);
36
37        db.salsa_event(Event {
38            runtime_id: runtime.id(),
39            kind: EventKind::WillExecute {
40                database_key: database_key_index,
41            },
42        });
43
44        // Query was not previously executed, or value is potentially
45        // stale, or value is absent. Let's execute!
46        let database_key_index = active_query.database_key_index;
47        let key = C::key_from_id(database_key_index.key_index);
48        let value = match Cycle::catch(|| C::execute(db, key)) {
49            Ok(v) => v,
50            Err(cycle) => {
51                log::debug!(
52                    "{:?}: caught cycle {:?}, have strategy {:?}",
53                    database_key_index.debug(db),
54                    cycle,
55                    C::CYCLE_STRATEGY
56                );
57                match C::CYCLE_STRATEGY {
58                    crate::cycle::CycleRecoveryStrategy::Panic => cycle.throw(),
59                    crate::cycle::CycleRecoveryStrategy::Fallback => {
60                        if let Some(c) = active_query.take_cycle() {
61                            assert!(c.is(&cycle));
62                            C::recover_from_cycle(db, &cycle, key)
63                        } else {
64                            // we are not a participant in this cycle
65                            debug_assert!(!cycle
66                                .participant_keys()
67                                .any(|k| k == database_key_index));
68                            cycle.throw()
69                        }
70                    }
71                }
72            }
73        };
74        let mut revisions = active_query.pop(runtime);
75
76        // We assume that query is side-effect free -- that is, does
77        // not mutate the "inputs" to the query system. Sanity check
78        // that assumption here, at least to the best of our ability.
79        assert_eq!(
80            runtime.current_revision(),
81            revision_now,
82            "revision altered during query execution",
83        );
84
85        // If the new value is equal to the old one, then it didn't
86        // really change, even if some of its inputs have. So we can
87        // "backdate" its `changed_at` revision to be the same as the
88        // old value.
89        if let Some(old_memo) = &opt_old_memo {
90            self.backdate_if_appropriate(old_memo, &mut revisions, &value);
91            self.diff_outputs(db, database_key_index, old_memo, &revisions);
92        }
93
94        let value = self
95            .insert_memo(
96                db,
97                key,
98                Memo::new(Some(value), revision_now, revisions.clone()),
99            )
100            .unwrap();
101
102        let stamped_value = revisions.stamped_value(value);
103
104        log::debug!(
105            "{:?}: read_upgrade: result.revisions = {:#?}",
106            database_key_index.debug(db),
107            revisions
108        );
109
110        stamped_value
111    }
112}
