32/// Function ingredients are the "workhorse" of salsa.
33/// They are used for tracked functions, for the "value" fields of tracked structs, and for the fields of input structs.
34/// The function ingredient is fairly complex and so its code is spread across multiple modules, typically one per method.
35/// The main entry points are:
36///
37/// * the `fetch` method, which is invoked when the function is called by the user's code;
38///   it will return a memoized value if one exists, or execute the function otherwise.
39/// * the `specify` method, which can only be used when the key is an entity created by the active query.
40///   It sets the value of the function imperatively, so that when later fetches occur, they'll return this value.
41/// * the `store` method, which can only be invoked with an `&mut` reference, and is to set input fields.
42pub struct FunctionIngredient<C: Configuration> {
43    /// The ingredient index we were assigned in the database.
44    /// Used to construct `DatabaseKeyIndex` values.
45    index: IngredientIndex,
46
47    /// Tracks the keys for which we have memoized values.
48    memo_map: memo::MemoMap<C::Key, C::Value>,
49
50    /// Tracks the keys that are currently being processed; used to coordinate between
51    /// worker threads.
52    sync_map: sync::SyncMap,
53
54    /// Used to find memos to throw out when we have too many memoized values.
55    lru: lru::Lru,
56
57    /// When `fetch` and friends executes, they return a reference to the
58    /// value stored in the memo that is extended to live as long as the `&self`
59    /// reference we start with. This means that whenever we remove something
60    /// from `memo_map` with an `&self` reference, there *could* be references to its
61    /// internals still in use. Therefore we push the memo into this queue and
62    /// only *actually* free up memory when a new revision starts (which means
63    /// we have an `&mut` reference to self).
64    ///
65    /// You might think that we could do this only if the memo was verified in the
66    /// current revision: you would be right, but we are being defensive, because
67    /// we don't know that we can trust the database to give us the same runtime
68    /// everytime and so forth.
69    deleted_entries: SegQueue<ArcSwap<memo::Memo<C::Value>>>,
70
71    /// Set to true once we invoke `register_dependent_fn` for `C::SalsaStruct`.
72    /// Prevents us from registering more than once.
73    registered: AtomicCell<bool>,
74
75    debug_name: &'static str,
76}

78pub trait Configuration {
79    type Jar: for<'db> Jar<'db>;
80
81    /// The "salsa struct type" that this function is associated with.
82    /// This can be just `salsa::Id` for functions that intern their arguments
83    /// and are not clearly associated with any one salsa struct.
84    type SalsaStruct: for<'db> SalsaStructInDb<DynDb<'db, Self>>;
85
86    /// What key is used to index the memo. Typically a salsa struct id,
87    /// but if this memoized function has multiple arguments it will be a `salsa::Id`
88    /// that results from interning those arguments.
89    type Key: AsId;
90
91    /// The value computed by the function.
92    type Value: fmt::Debug;
93
94    /// Determines whether this function can recover from being a participant in a cycle
95    /// (and, if so, how).
96    const CYCLE_STRATEGY: CycleRecoveryStrategy;
97
98    /// Invokes after a new result `new_value`` has been computed for which an older memoized
99    /// value existed `old_value`. Returns true if the new value is equal to the older one
100    /// and hence should be "backdated" (i.e., marked as having last changed in an older revision,
101    /// even though it was recomputed).
102    ///
103    /// This invokes user's code in form of the `Eq` impl.
104    fn should_backdate_value(old_value: &Self::Value, new_value: &Self::Value) -> bool;
105
106    /// Invoked when we need to compute the value for the given key, either because we've never
107    /// computed it before or because the old one relied on inputs that have changed.
108    ///
109    /// This invokes the function the user wrote.
110    fn execute(db: &DynDb<Self>, key: Self::Key) -> Self::Value;
111
112    /// If the cycle strategy is `Recover`, then invoked when `key` is a participant
113    /// in a cycle to find out what value it should have.
114    ///
115    /// This invokes the recovery function given by the user.
116    fn recover_from_cycle(db: &DynDb<Self>, cycle: &Cycle, key: Self::Key) -> Self::Value;
117
118    /// Given a salsa Id, returns the key. Convenience function to avoid
119    /// having to type `<C::Key as AsId>::from_id`.
120    fn key_from_id(id: Id) -> Self::Key {
121        AsId::from_id(id)
122    }
123}
124
125/// True if `old_value == new_value`. Invoked by the generated
126/// code for `should_backdate_value` so as to give a better
127/// error message.
128pub fn should_backdate_value<V: Eq>(old_value: &V, new_value: &V) -> bool {
129    old_value == new_value
130}

138impl<C> FunctionIngredient<C>
139where
140    C: Configuration,
141{
142    pub fn new(index: IngredientIndex, debug_name: &'static str) -> Self {
143        Self {
144            index,
145            memo_map: memo::MemoMap::default(),
146            lru: Default::default(),
147            sync_map: Default::default(),
148            deleted_entries: Default::default(),
149            registered: Default::default(),
150            debug_name,
151        }
152    }
153
154    fn database_key_index(&self, k: C::Key) -> DatabaseKeyIndex {
155        DatabaseKeyIndex {
156            ingredient_index: self.index,
157            key_index: k.as_id(),
158        }
159    }
160
161    pub fn set_capacity(&self, capacity: usize) {
162        self.lru.set_capacity(capacity);
163    }
164
165    /// Returns a reference to the memo value that lives as long as self.
166    /// This is UNSAFE: the caller is responsible for ensuring that the
167    /// memo will not be released so long as the `&self` is valid.
168    /// This is done by (a) ensuring the memo is present in the memo-map
169    /// when this function is called and (b) ensuring that any entries
170    /// removed from the memo-map are added to `deleted_entries`, which is
171    /// only cleared with `&mut self`.
172    unsafe fn extend_memo_lifetime<'this, 'memo>(
173        &'this self,
174        memo: &'memo memo::Memo<C::Value>,
175    ) -> Option<&'this C::Value> {
176        let memo_value: Option<&'memo C::Value> = memo.value.as_ref();
177        std::mem::transmute(memo_value)
178    }
179
180    fn insert_memo(
181        &self,
182        db: &DynDb<'_, C>,
183        key: C::Key,
184        memo: memo::Memo<C::Value>,
185    ) -> Option<&C::Value> {
186        self.register(db);
187        let memo = Arc::new(memo);
188        let value = unsafe {
189            // Unsafety conditions: memo must be in the map (it's not yet, but it will be by the time this
190            // value is returned) and anything removed from map is added to deleted entries (ensured elsewhere).
191            self.extend_memo_lifetime(&memo)
192        };
193        if let Some(old_value) = self.memo_map.insert(key, memo) {
194            // In case there is a reference to the old memo out there, we have to store it
195            // in the deleted entries. This will get cleared when a new revision starts.
196            self.deleted_entries.push(old_value);
197        }
198        value
199    }
200
201    /// Register this function as a dependent fn of the given salsa struct.
202    /// When instances of that salsa struct are deleted, we'll get a callback
203    /// so we can remove any data keyed by them.
204    fn register(&self, db: &DynDb<'_, C>) {
205        if !self.registered.fetch_or(true) {
206            <C::SalsaStruct as SalsaStructInDb<_>>::register_dependent_fn(db, self.index)
207        }
208    }
209}

211impl<DB, C> Ingredient<DB> for FunctionIngredient<C>
212where
213    DB: ?Sized + DbWithJar<C::Jar>,
214    C: Configuration,
215{
216    fn ingredient_index(&self) -> IngredientIndex {
217        self.index
218    }
219
220    fn maybe_changed_after(&self, db: &DB, input: DependencyIndex, revision: Revision) -> bool {
221        let key = C::key_from_id(input.key_index.unwrap());
222        let db = db.as_jar_db();
223        self.maybe_changed_after(db, key, revision)
224    }
225
226    fn cycle_recovery_strategy(&self) -> CycleRecoveryStrategy {
227        C::CYCLE_STRATEGY
228    }
229
230    fn origin(&self, key_index: Id) -> Option<QueryOrigin> {
231        let key = C::key_from_id(key_index);
232        self.origin(key)
233    }
234
235    fn mark_validated_output(
236        &self,
237        db: &DB,
238        executor: DatabaseKeyIndex,
239        output_key: Option<crate::Id>,
240    ) {
241        let output_key = C::key_from_id(output_key.unwrap());
242        self.validate_specified_value(db.as_jar_db(), executor, output_key);
243    }
244
245    fn remove_stale_output(
246        &self,
247        _db: &DB,
248        _executor: DatabaseKeyIndex,
249        _stale_output_key: Option<crate::Id>,
250    ) {
251        // This function is invoked when a query Q specifies the value for `stale_output_key` in rev 1,
252        // but not in rev 2. We don't do anything in this case, we just leave the (now stale) memo.
253        // Since its `verified_at` field has not changed, it will be considered dirty if it is invoked.
254    }
255
256    fn reset_for_new_revision(&mut self) {
257        std::mem::take(&mut self.deleted_entries);
258    }
259
260    fn salsa_struct_deleted(&self, db: &DB, id: crate::Id) {
261        // Remove any data keyed by `id`, since `id` no longer
262        // exists in this revision.
263
264        let id: C::Key = C::key_from_id(id);
265        if let Some(origin) = self.delete_memo(id) {
266            let key = self.database_key_index(id);
267            db.salsa_event(Event {
268                runtime_id: db.runtime().id(),
269                kind: EventKind::DidDiscard { key },
270            });
271
272            // Anything that was output by this memoized execution
273            // is now itself stale.
274            for stale_output in origin.outputs() {
275                db.remove_stale_output(key, stale_output)
276            }
277        }
278    }
279
280    fn fmt_index(&self, index: Option<crate::Id>, fmt: &mut fmt::Formatter<'_>) -> fmt::Result {
281        fmt_index(self.debug_name, index, fmt)
282    }
283}
